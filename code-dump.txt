const CACHE_PREFIX = `pb-weather:${self.registration.scope}:`;
const CACHE_NAME = `${CACHE_PREFIX}v4`;
const INDEX_URL = new URL('index.html', self.registration.scope).href;
const MANIFEST_URL = new URL('manifest.json?v=2', self.registration.scope).href;

self.addEventListener('install', (event) => {
  event.waitUntil((async () => {
    const cache = await caches.open(CACHE_NAME);
    await cache.addAll(['./', './index.html', './manifest.json', MANIFEST_URL, './icon-192.png', './icon-512.png', './favicon.svg']);
    // The build manifest includes lazy chunks too, so every tab works on the first offline launch.
    const response = await fetch(new URL('asset-manifest.json', self.registration.scope), { cache: 'no-cache' });
    if (!response.ok) throw new Error('Could not load the offline asset manifest');
    const manifest = await response.json();
    const assets = Object.values(manifest).flatMap((entry) => [entry.file, ...(entry.css ?? []), ...(entry.assets ?? [])]);
    await cache.addAll([...new Set(assets)].map((asset) => new URL(asset, INDEX_URL).href));
    await self.skipWaiting();
  })());
});

self.addEventListener('activate', (event) => {
  event.waitUntil((async () => {
    const keys = await caches.keys();
    await Promise.all(keys.filter((key) => key.startsWith(CACHE_PREFIX) && key !== CACHE_NAME)
      .map((key) => caches.delete(key)));
    await self.clients.claim();
  })());
});

self.addEventListener('fetch', (event) => {
  const url = new URL(event.request.url);
  const scope = new URL(self.registration.scope);
  // Weather data is cached by the app in IndexedDB.
  if (event.request.method !== 'GET' || url.origin !== scope.origin || !url.pathname.startsWith(scope.pathname)) return;

  if (event.request.mode === 'navigate' || url.pathname === new URL(MANIFEST_URL).pathname) {
    // Fetch current screens and install settings, retaining offline copies.
    const cacheKey = event.request.mode === 'navigate' ? INDEX_URL : event.request;
    event.respondWith((async () => {
      const cache = await caches.open(CACHE_NAME);
      try {
        const response = await fetch(event.request, { cache: 'no-cache' });
        if (response.ok) {
          await cache.put(cacheKey, response.clone());
          return response;
        }
        return (await cache.match(cacheKey)) || response;
      } catch {
        return (await cache.match(cacheKey)) || Response.error();
      }
    })());
    return;
  }

  event.respondWith((async () => {
    const cache = await caches.open(CACHE_NAME);
    // These files are static. CORS module requests can carry a different Origin header
    // from precache requests, but still need the same saved file when offline.
    const cached = await cache.match(event.request, { ignoreVary: true });
    if (cached) return cached;
    try {
      const response = await fetch(event.request);
      if (response.ok) {
        await cache.put(event.request, response.clone());
      }
      return response;
    } catch {
      return Response.error();
    }
  })());
});

/**
 * Aurora / geomagnetic data from NOAA Space Weather Prediction Center (SWPC).
 *
 * At Peterborough (52.57°N), aurora is typically visible when Kp ≥ 7.
 * Kp 5–6 means aurora is possible from Scotland/Northern England.
 * Kp 8+ means a good chance of aurora across central England.
 */

const CACHE_KEY = 'pb-weather:aurora:v2';
const CACHE_TTL = 30 * 60_000; // 30 minutes

export interface KpForecastEntry {
  time: string;    // UTC time string
  kp: number;      // Predicted Kp index (0–9)
  observed: boolean;
}

export interface AuroraData {
  current: { kp: number; time: string } | null;
  forecast: KpForecastEntry[];
  maxKp3Day: number;
  /** Visibility assessment for Peterborough */
  visibility: 'None expected' | 'Unlikely' | 'Possible' | 'Good chance' | 'Strong display likely';
  /** Brief explanation */
  visibilityNote: string;
  fetchedAt: number;
}

/** Kp threshold for Peterborough at 52.57°N geomagnetic latitude */
function assessVisibility(maxKp: number): Pick<AuroraData, 'visibility' | 'visibilityNote'> {
  // Geomagnetic latitude of Peterborough is roughly 54.5°N (higher than geographic)
  if (maxKp >= 9) return {
    visibility: 'Strong display likely',
    visibilityNote: `Kp ${maxKp} — a severe storm. Aurora should be visible across England, potentially to the south. Look north for curtains of light.`,
  };
  if (maxKp >= 8) return {
    visibility: 'Good chance',
    visibilityNote: `Kp ${maxKp} — a strong storm. Aurora may be visible low on the northern horizon from Peterborough. Find a dark site with a clear view north.`,
  };
  if (maxKp >= 7) return {
    visibility: 'Possible',
    visibilityNote: `Kp ${maxKp} — aurora may reach central England. Check for a faint glow or green tint low on the northern horizon after dark.`,
  };
  if (maxKp >= 5) return {
    visibility: 'Unlikely',
    visibilityNote: `Kp ${maxKp} — aurora is active at higher latitudes. Unlikely from Peterborough but possible from Scotland or northern England.`,
  };
  return {
    visibility: 'None expected',
    visibilityNote: 'Geomagnetic activity is quiet. Aurora is not expected at Peterborough\'s latitude.',
  };
}

/** Parse the NOAA Kp forecast JSON */
function parseForecast(raw: Array<{ time_tag: string; kp: number; observed: string }>): KpForecastEntry[] {
  return raw.map((entry) => ({
    time: entry.time_tag,
    kp: entry.kp ?? 0,
    observed: entry.observed === 'observed',
  }));
}

/** Fetch the latest 1-minute Kp observation */
async function fetchCurrentKp(): Promise<{ kp: number; time: string } | null> {
  try {
    const res = await fetch('https://services.swpc.noaa.gov/json/planetary_k_index_1m.json', { signal: AbortSignal.timeout(15_000) });
    if (!res.ok) return null;
    const data = await res.json() as Array<{ time_tag: string; kp_index: number }>;
    if (!data.length) return null;
    const latest = data[data.length - 1];
    return { kp: latest.kp_index, time: latest.time_tag };
  } catch {
    return null;
  }
}

/** Fetch 3-day Kp forecast from NOAA SWPC */
async function fetchKpForecast(): Promise<KpForecastEntry[]> {
  try {
    const res = await fetch('https://services.swpc.noaa.gov/products/noaa-planetary-k-index-forecast.json', { signal: AbortSignal.timeout(15_000) });
    if (!res.ok) return [];
    const data = await res.json();
    return parseForecast(data);
  } catch {
    return [];
  }
}

/** NOAA includes historical observations before its predictions. Keep the current and upcoming blocks. */
export function upcomingKpForecast(entries: KpForecastEntry[], now = Date.now()): KpForecastEntry[] {
  return entries.filter((entry) => {
    const time = Date.parse(entry.time.endsWith('Z') ? entry.time : `${entry.time}Z`);
    return !entry.observed && Number.isFinite(entry.kp) && entry.kp >= 0 && entry.kp <= 9
      && time + 3 * 3600_000 > now && time < now + 72 * 3600_000;
  }).sort((a, b) => a.time.localeCompare(b.time));
}

/** Fetch aurora data with localStorage caching */
export async function fetchAuroraData(force = false): Promise<AuroraData | null> {
  // Check cache
  try {
    const cached = localStorage.getItem(CACHE_KEY);
    if (!force && cached) {
      const parsed = JSON.parse(cached) as AuroraData;
      if (Date.now() - parsed.fetchedAt < CACHE_TTL) return parsed;
    }
  } catch { /* ignore corrupt cache */ }

  const [current, history] = await Promise.all([fetchCurrentKp(), fetchKpForecast()]);

  const forecast = upcomingKpForecast(history);
  if (!current && forecast.length === 0) return null;

  const allKp = [...forecast.map((f) => f.kp), ...(current ? [current.kp] : [])];
  const maxKp3Day = Math.max(...allKp, 0);
  const { visibility, visibilityNote } = assessVisibility(maxKp3Day);

  const result: AuroraData = {
    current,
    forecast,
    maxKp3Day,
    visibility,
    visibilityNote,
    fetchedAt: Date.now(),
  };

  try {
    localStorage.setItem(CACHE_KEY, JSON.stringify(result));
  } catch { /* storage full */ }

  return result;
}

/** Kp level colour for UI */
export function kpColour(kp: number): string {
  if (kp >= 8) return '#ef6b73';
  if (kp >= 6) return '#f2bb9d';
  if (kp >= 4) return '#eed49d';
  if (kp >= 2) return '#b5dacd';
  return '#a6b2c5';
}

/** Kp level label */
export function kpLabel(kp: number): string {
  if (kp >= 8) return 'Severe storm';
  if (kp >= 7) return 'Strong storm';
  if (kp >= 6) return 'Moderate storm';
  if (kp >= 5) return 'Minor storm';
  if (kp >= 4) return 'Active';
  if (kp >= 2) return 'Unsettled';
  return 'Quiet';
}

import { openDB, type IDBPDatabase } from 'idb';
import type { CachedRun } from '../types';
import { RUN_LOOKBACK_HOURS, MS_PER_HOUR } from '../config/models';

const DB_NAME = 'weather-trends';
const DB_VERSION = 3; // v2: wind units fixed (km/h → m/s); v3: unused ensemble store dropped

let dbPromise: Promise<IDBPDatabase> | null = null;

function getDb() {
  if (!dbPromise) {
    dbPromise = openDB(DB_NAME, DB_VERSION, {
      upgrade(db, oldVersion) {
        // Wipe stale stores on version bump (v1 had wrong wind units)
        if (oldVersion < 2) {
          if (db.objectStoreNames.contains('runs')) db.deleteObjectStore('runs');
        }
        if (!db.objectStoreNames.contains('runs')) {
          const store = db.createObjectStore('runs', { keyPath: 'key' });
          store.createIndex('by-model', 'modelId');
          store.createIndex('by-fetchedAt', 'fetchedAt');
        }
        if (db.objectStoreNames.contains('ensembles')) db.deleteObjectStore('ensembles');
      },
    });
  }
  return dbPromise;
}

export async function putCachedRun(run: CachedRun): Promise<void> {
  const db = await getDb();
  await db.put('runs', run);
}

/** Store several records in one transaction (one model run = one record per variable). */
export async function putCachedRuns(runs: CachedRun[]): Promise<void> {
  const db = await getDb();
  const tx = db.transaction('runs', 'readwrite');
  await Promise.all([...runs.map((run) => tx.store.put(run)), tx.done]);
}

export async function getAllCachedRuns(): Promise<CachedRun[]> {
  const db = await getDb();
  return db.getAll('runs');
}

export async function pruneOldRuns(): Promise<void> {
  const db = await getDb();
  const cutoff = Date.now() - RUN_LOOKBACK_HOURS * MS_PER_HOUR;
  // Keys are `${modelId}__${runTime}__${variable}`, so no need to load every forecast array.
  const keys = (await db.getAllKeys('runs')) as string[];
  const stale = keys.filter((key) => {
    const runTime = key.split('__')[1];
    return !runTime || Date.parse(`${runTime}Z`) < cutoff;
  });
  if (!stale.length) return;
  const tx = db.transaction('runs', 'readwrite');
  await Promise.all([...stale.map((key) => tx.store.delete(key)), tx.done]);
}

import { LOCATION } from '../config/models';
import type { VariableId } from '../config/models';
import type { CachedRun } from '../types';
import { putCachedRuns, pruneOldRuns } from './db';
import { BASE_VARIABLES, getRunTasks, makeRunKey } from './runSchedule';

const SINGLE_RUNS_BASE = 'https://single-runs-api.open-meteo.com/v1/forecast';

type FetchResult = 'ok' | 'unavailable' | 'network';

/**
 * Fetch every variable for one model run in a single request (Open-Meteo counts a request
 * with up to 10 variables and 7 days as one call), then store one record per variable.
 */
async function fetchModelRun(modelId: string, runTime: string, variables: VariableId[]): Promise<FetchResult> {
  const params = new URLSearchParams({
    latitude: String(LOCATION.latitude),
    longitude: String(LOCATION.longitude),
    hourly: variables.join(','),
    models: modelId,
    run: runTime,
    forecast_days: '7',
    timezone: 'UTC',
    wind_speed_unit: 'ms',
  });

  let res: Response;
  try {
    res = await fetch(`${SINGLE_RUNS_BASE}?${params}`, { signal: AbortSignal.timeout(20_000) });
  } catch {
    return 'network';
  }
  if (res.status === 429 || res.status >= 500) return 'network';
  if (!res.ok) return 'unavailable'; // run not published yet, or not offered for this model

  try {
    const hourly = (await res.json())?.hourly;
    if (!hourly || !Array.isArray(hourly.time)) return 'unavailable';
    const fetchedAt = Date.now();
    const records: CachedRun[] = variables
      .filter((variable) => Array.isArray(hourly[variable]) && hourly[variable].some((v: unknown) => typeof v === 'number'))
      .map((variable) => ({
        key: makeRunKey(modelId, runTime, variable),
        modelId, runTime, variable,
        times: hourly.time,
        values: hourly[variable],
        fetchedAt,
      }));
    if (!records.length) return 'unavailable';
    await putCachedRuns(records);
    return 'ok';
  } catch {
    return 'network';
  }
}

export interface FetchSummary {
  requested: number;
  ok: number;
  unavailable: number;
  failed: number;
}

/** Fetch new or still-publishing runs for all deterministic models. */
export async function fetchAllDeterministicRuns(
  cached: CachedRun[],
  onProgress?: (done: number, total: number) => void
): Promise<FetchSummary> {
  await pruneOldRuns();

  const tasks = getRunTasks(cached);

  const summary: FetchSummary = { requested: tasks.length, ok: 0, unavailable: 0, failed: 0 };
  let done = 0;
  onProgress?.(0, tasks.length);

  // ~120 requests on a cold start, a handful afterwards; stay well under 600/min.
  const BATCH_SIZE = 6;
  for (let i = 0; i < tasks.length; i += BATCH_SIZE) {
    const batch = tasks.slice(i, i + BATCH_SIZE);
    await Promise.all(batch.map(async (t) => {
      let result = await fetchModelRun(t.modelId, t.runTime, t.variables);
      // If a model rejects one of the optional variables, don't lose its core data.
      if (result === 'unavailable' && t.variables.length > BASE_VARIABLES.length) {
        result = await fetchModelRun(t.modelId, t.runTime, BASE_VARIABLES);
      }
      if (result === 'ok') summary.ok++;
      else if (result === 'unavailable') summary.unavailable++;
      else summary.failed++;
      onProgress?.(++done, tasks.length);
    }));
    if (i + BATCH_SIZE < tasks.length) await new Promise((r) => setTimeout(r, 200));
  }
  return summary;
}

export { makeRunKey };

import { LOCATION } from '../config/models';
import type { MonthlyOutlookData } from '../types';

const CACHE_KEY = 'pb-weather:monthly-outlook:v1';
const CACHE_AGE = 6 * 3600_000;
let pending: Promise<MonthlyOutlookData> | null = null;

export function parseMonthlyOutlook(response: unknown, fetchedAt = Date.now()): MonthlyOutlookData {
  const weekly = (response as { weekly?: Record<string, unknown> } | null)?.weekly;
  if (!weekly || !Array.isArray(weekly.time)) throw new Error('Long-range data unavailable');
  const value = (name: string, index: number) => {
    const values = weekly[name];
    const item = Array.isArray(values) ? values[index] : null;
    return typeof item === 'number' && Number.isFinite(item) ? item : null;
  };
  const weeks = weekly.time.flatMap((date: unknown, index: number) => {
    if (typeof date !== 'string' || !/^\d{4}-\d{2}-\d{2}$/.test(date) || !Number.isFinite(Date.parse(`${date}T00:00:00Z`)) || new Date(`${date}T00:00:00Z`).toISOString().slice(0, 10) !== date) return [];
    const rainfall = value('precipitation_mean', index);
    return [{ date, temperatureAnomaly: value('temperature_2m_anomaly', index),
      rainfallMm: rainfall !== null && rainfall >= 0 ? rainfall : null,
      rainfallAnomaly: value('precipitation_anomaly', index) }];
  }).sort((a, b) => a.date.localeCompare(b.date));
  if (!weeks.some((week) => week.temperatureAnomaly !== null || week.rainfallMm !== null)) throw new Error('Long-range data unavailable');
  return { fetchedAt, weeks: weeks.filter((week, index) => index === 0 || week.date !== weeks[index - 1].date) };
}

export function readMonthlyOutlook(): MonthlyOutlookData | null {
  try {
    const saved = JSON.parse(localStorage.getItem(CACHE_KEY) ?? 'null');
    if (!saved || typeof saved.fetchedAt !== 'number' || !Number.isFinite(saved.fetchedAt) || saved.fetchedAt < 0 || saved.fetchedAt > Date.now() + 60_000) return null;
    return parseMonthlyOutlook(saved.response, saved.fetchedAt);
  } catch { return null; }
}

export function fetchMonthlyOutlook(today: string, force = false): Promise<MonthlyOutlookData> {
  const cached = readMonthlyOutlook();
  if (!force && cached && Date.now() - cached.fetchedAt < CACHE_AGE && cached.weeks.filter((week) => week.date >= today).length >= 4) return Promise.resolve(cached);
  if (pending) return pending;
  pending = (async () => {
    const params = new URLSearchParams({ latitude: String(LOCATION.latitude), longitude: String(LOCATION.longitude),
      models: 'ecmwf_ec46', weekly: 'temperature_2m_anomaly,precipitation_mean,precipitation_anomaly',
      forecast_days: '35', timezone: 'Europe/London' });
    const response = await fetch(`https://seasonal-api.open-meteo.com/v1/seasonal?${params}`, { signal: AbortSignal.timeout(15_000) });
    if (!response.ok) throw new Error('Long-range data unavailable');
    const data: unknown = await response.json();
    const outlook = parseMonthlyOutlook(data);
    if (!outlook.weeks.some((week) => week.date >= today)) throw new Error('No upcoming long-range data');
    try { localStorage.setItem(CACHE_KEY, JSON.stringify({ fetchedAt: outlook.fetchedAt, response: data })); } catch { /* Reading forecasts still works when storage is unavailable. */ }
    return outlook;
  })().finally(() => { pending = null; });
  return pending;
}

import { DETERMINISTIC_MODELS, VARIABLES, RUN_LOOKBACK_HOURS, MS_PER_HOUR } from '../config/models.ts';
import type { VariableId } from '../config/models.ts';
import type { CachedRun } from '../types.ts';

// Runs fetched this soon after their cycle time may still be publishing.
const PROVISIONAL_HOURS = 9;
// Avoid immediately retrying missing or still-publishing variables on every refresh.
const RECHECK_MS = 30 * 60_000;
export const BASE_VARIABLES: VariableId[] = VARIABLES.map((v) => v.id);
const EXTRA_VARIABLES: VariableId[] = ['snowfall', 'weather_code', 'is_day', 'cloud_cover'];
const EXTRA_RUNS = 4;

export function makeRunKey(modelId: string, runTime: string, variable: VariableId): string {
  return `${modelId}__${runTime}__${variable}`;
}

function generateRunTimes(cycles: number[], now: number): string[] {
  const cutoff = now - RUN_LOOKBACK_HOURS * MS_PER_HOUR;
  const times: string[] = [];
  const nowDate = new Date(now);
  const todayUTC = Date.UTC(nowDate.getUTCFullYear(), nowDate.getUTCMonth(), nowDate.getUTCDate());

  for (let dayOffset = 0; dayOffset <= Math.ceil(RUN_LOOKBACK_HOURS / 24) + 1; dayOffset++) {
    for (const cycle of cycles) {
      const runMs = todayUTC - dayOffset * 24 * MS_PER_HOUR + cycle * MS_PER_HOUR;
      if (runMs <= now && runMs >= cutoff) times.push(new Date(runMs).toISOString().slice(0, 16));
    }
  }
  return [...new Set(times)].sort();
}

/** Plan requests from the cache, retrying incomplete runs after the recheck interval. */
export function getRunTasks(cached: CachedRun[], now = Date.now()) {
  const fetchedAt = new Map(cached.map((run) => [run.key, run.fetchedAt]));
  const tasks: { modelId: string; runTime: string; variables: VariableId[] }[] = [];

  for (const model of DETERMINISTIC_MODELS) {
    const runTimes = generateRunTimes(model.runCycles, now);
    const recent = new Set(runTimes.slice(-EXTRA_RUNS));
    for (const runTime of runTimes) {
      const wantsExtras = !model.isOverlay && recent.has(runTime);
      const variables = wantsExtras ? [...BASE_VARIABLES, ...EXTRA_VARIABLES] : BASE_VARIABLES;
      const anchor = fetchedAt.get(makeRunKey(model.id, runTime, 'temperature_2m'));
      const recentlyChecked = anchor !== undefined && now - anchor < RECHECK_MS;
      const provisional = anchor !== undefined
        && anchor - Date.parse(`${runTime}Z`) < PROVISIONAL_HOURS * MS_PER_HOUR;
      // A temperature record alone does not guarantee the other variables were saved.
      const missingVariables = variables.some((variable) => !fetchedAt.has(makeRunKey(model.id, runTime, variable)));

      if (anchor === undefined || ((provisional || missingVariables) && !recentlyChecked)) {
        tasks.push({ modelId: model.id, runTime, variables });
      }
    }
  }
  return tasks;
}

/** Seasonal SEAS5 monthly outlook and EC46 weekly comparison from Open-Meteo. */

import { LOCATION } from '../config/models.ts';

const CACHE_KEY = 'pb-weather:seasonal:v2';
const CACHE_TTL = 6 * 3600_000; // 6 hours

export interface SeasonalModelConfig {
  id: string;

  agency: string;
  colour: string;

}

export const SEASONAL_MODELS: SeasonalModelConfig[] = [
  { id: 'ecmwf_seas5', label: 'ECMWF SEAS5', agency: 'ECMWF', colour: '#c8c2ec', hasEnsemble: true },
  { id: 'ecmwf_ec46', label: 'ECMWF EC46', agency: 'ECMWF', colour: '#85c9c1', hasEnsemble: true },

export interface SeasonalWeek {
  date: string; // ISO week start
  tempAnomaly: number | null;
  precipMm: number | null;
  precipAnomaly: number | null;
}

export interface SeasonalModelData {
  modelId: string;
  weeks: SeasonalWeek[];
}

export interface SeasonalMonth {
  /** YYYY-MM */
  month: string;
  label: string;
  /** Temperature anomaly per model (modelId → °C) */
  tempAnomalies: Map<string, number>;
  /** Rainfall anomaly per model (modelId → mm/week) */
  precipAnomalies: Map<string, number>;
  /** Mean weekly rainfall per model (modelId → mm/week) */
  precipTotals: Map<string, number>;
  /** Multi-model mean temp anomaly */
  tempMean: number | null;
  /** Multi-model mean precip anomaly */
  precipMean: number | null;
  /** Model agreement on temperature direction */
  tempAgreement: 'strong' | 'moderate' | 'weak' | 'none';
  /** Model agreement on precip direction */
  precipAgreement: 'strong' | 'moderate' | 'weak' | 'none';
}

export interface SeasonalData {
  models: SeasonalModelData[];
  months: SeasonalMonth[];
  fetchedAt: number;
}

/** Fetch weekly seasonal data for one model */
async function fetchModel(modelId: string): Promise<SeasonalModelData | null> {
  const monthly = modelId === 'ecmwf_seas5';
  const params = new URLSearchParams({
    latitude: String(LOCATION.latitude),
    longitude: String(LOCATION.longitude),
    models: modelId,
    [monthly ? 'monthly' : 'weekly']: 'temperature_2m_anomaly,precipitation_mean,precipitation_anomaly',
    forecast_days: monthly ? '210' : '46',
    timezone: 'Europe/London',
  });
  try {
    const res = await fetch(`https://seasonal-api.open-meteo.com/v1/seasonal?${params}`, {
      signal: AbortSignal.timeout(15_000),
    });
    if (!res.ok) return null;
    const data = await res.json();
    const weekly = data?.[monthly ? 'monthly' : 'weekly'];
    if (!weekly || !Array.isArray(weekly.time)) return null;

    const weeks: SeasonalWeek[] = [];
    for (let i = 0; i < weekly.time.length; i++) {
      const date = weekly.time[i];
      if (typeof date !== 'string' || !/^\d{4}-\d{2}-\d{2}$/.test(date) || !Number.isFinite(Date.parse(date))) continue;
      const tempAnomaly = safeNum(weekly.temperature_2m_anomaly?.[i]);
      // Present both systems on a common mm/week scale. SEAS5 supplies monthly totals.
      const days = new Date(Date.UTC(Number(date.slice(0, 4)), Number(date.slice(5, 7)), 0)).getUTCDate();
      const scale = monthly ? 7 / days : 1;
      const rainfall = safeNum(weekly.precipitation_mean?.[i]);
      const anomaly = safeNum(weekly.precipitation_anomaly?.[i]);
      const precipMm = rainfall === null ? null : rainfall * scale;
      const precipAnomaly = anomaly === null ? null : anomaly * scale;
      if (tempAnomaly === null && precipMm === null && precipAnomaly === null) continue;
      weeks.push({ date, tempAnomaly, precipMm, precipAnomaly });
    }

    return weeks.length ? { modelId, weeks } : null;
  } catch {
    return null;
  }
}

function safeNum(v: unknown): number | null {
  return typeof v === 'number' && Number.isFinite(v) ? v : null;
}

/** Aggregate weekly data into monthly buckets and compute multi-model stats */
function aggregateMonths(models: SeasonalModelData[], today: string): SeasonalMonth[] {
  // Collect all unique months across all models
  const monthMap = new Map<string, Map<string, { temps: number[]; precips: number[]; precipTotals: number[] }>>();

  for (const model of models) {
    for (const week of model.weeks) {
      if (model.modelId === 'ecmwf_seas5' ? week.date.slice(0, 7) < today.slice(0, 7) : week.date < today) continue;
      const ym = week.date.slice(0, 7); // YYYY-MM
      if (!monthMap.has(ym)) monthMap.set(ym, new Map());
      const modelMap = monthMap.get(ym)!;
      if (!modelMap.has(model.modelId)) modelMap.set(model.modelId, { temps: [], precips: [], precipTotals: [] });
      const bucket = modelMap.get(model.modelId)!;
      if (week.tempAnomaly !== null) bucket.temps.push(week.tempAnomaly);
      if (week.precipAnomaly !== null) bucket.precips.push(week.precipAnomaly);
      if (week.precipMm !== null) bucket.precipTotals.push(week.precipMm);
    }
  }

  const months: SeasonalMonth[] = [];
  const sortedKeys = [...monthMap.keys()].sort();

  for (const ym of sortedKeys) {
    const modelData = monthMap.get(ym)!;
    const tempAnomalies = new Map<string, number>();
    const precipAnomalies = new Map<string, number>();
    const precipTotals = new Map<string, number>();

    for (const [modelId, bucket] of modelData) {
      if (bucket.temps.length > 0) {
        tempAnomalies.set(modelId, avg(bucket.temps));
      }
      if (bucket.precips.length > 0) {
        precipAnomalies.set(modelId, avg(bucket.precips));
      }
      if (bucket.precipTotals.length > 0) {
        precipTotals.set(modelId, avg(bucket.precipTotals));
      }
    }

    const tempValues = [...tempAnomalies.values()];
    const precipValues = [...precipAnomalies.values()];

    const date = new Date(`${ym}-15T12:00:00Z`);
    const label = date.toLocaleDateString('en-GB', { month: 'long', year: 'numeric', timeZone: 'Europe/London' });

    months.push({
      month: ym,
      label,
      tempAnomalies,
      precipAnomalies,
      precipTotals,
      tempMean: tempValues.length > 0 ? avg(tempValues) : null,
      precipMean: precipValues.length > 0 ? avg(precipValues) : null,
      tempAgreement: calcAgreement(tempValues),
      precipAgreement: calcAgreement(precipValues),
    });
  }

  return months;
}

function avg(values: number[]): number {
  return values.reduce((a, b) => a + b, 0) / values.length;
}

/** How many models agree on the sign of the anomaly? */
function calcAgreement(values: number[]): 'strong' | 'moderate' | 'weak' | 'none' {
  if (values.length < 2) return 'none';
  const positive = values.filter((v) => v > 0.1).length;
  const negative = values.filter((v) => v < -0.1).length;
  const total = values.length;
  const dominant = Math.max(positive, negative);
  const ratio = dominant / total;
  if (ratio >= 0.8) return 'strong';
  if (ratio >= 0.6) return 'moderate';
  if (ratio >= 0.4) return 'weak';
  return 'none';
}

/** Fetch all seasonal models and aggregate */
export async function fetchSeasonalData(today: string, force = false): Promise<SeasonalData> {
  // Check cache
  try {
    const cached = localStorage.getItem(CACHE_KEY);
    if (!force && cached) {
      const parsed = JSON.parse(cached) as { fetchedAt: number; raw: SeasonalModelData[] };
      if (Array.isArray(parsed.raw) && Date.now() - parsed.fetchedAt < CACHE_TTL) {
        const months = aggregateMonths(parsed.raw, today);
        return { models: parsed.raw, months, fetchedAt: parsed.fetchedAt };
      }
    }
  } catch { /* corrupt cache */ }

  // Fetch all models in parallel
  const results = await Promise.allSettled(
    SEASONAL_MODELS.map((m) => fetchModel(m.id))
  );

  const models: SeasonalModelData[] = [];
  for (const r of results) {
    if (r.status === 'fulfilled' && r.value) models.push(r.value);
  }

  if (models.length === 0) throw new Error('No seasonal data available');

  const months = aggregateMonths(models, today);
  if (!months.length) throw new Error('No upcoming seasonal data');
  const fetchedAt = Date.now();

  // Cache the raw model data
  try {
    localStorage.setItem(CACHE_KEY, JSON.stringify({ fetchedAt, raw: models }));
  } catch { /* storage full */ }

  return { models, months, fetchedAt };
}

/** Human-readable temperature anomaly */
export function tempSignal(anomaly: number | null): string {
  if (anomaly === null) return 'Uncertain';
  if (anomaly >= 1.5) return 'Much warmer';
  if (anomaly >= 0.5) return 'Warmer';
  if (anomaly <= -1.5) return 'Much cooler';
  if (anomaly <= -0.5) return 'Cooler';
  return 'Near average';
}

/** Human-readable precipitation anomaly */
export function precipSignal(anomaly: number | null): string {
  if (anomaly === null) return 'Uncertain';
  if (anomaly >= 5) return 'Much wetter';
  if (anomaly >= 1) return 'Wetter';
  if (anomaly <= -5) return 'Much drier';
  if (anomaly <= -1) return 'Drier';
  return 'Near average';
}

const API_BASE = 'https://enso-teleconnections.christopherwcorwin.workers.dev/api/teleconnections';
const CACHE_KEY = 'pb-weather:teleconnections:v2';
const CACHE_AGE = 6 * 3600_000; // 6 hours

export interface TeleconnectionIndex {
  slug: string;
  name: string;
  value: number;
  date: string;
  trend?: 'rising' | 'falling' | 'steady';
}

export interface MJOData {
  phase: number;
  amplitude: number;
  rmm1: number;
  rmm2: number;
  date: string;
}

export interface TeleconnectionSnapshot {
  nao: TeleconnectionIndex | null;
  ao: TeleconnectionIndex | null;
  enso: TeleconnectionIndex | null;
  pna: TeleconnectionIndex | null;
  ea: TeleconnectionIndex | null;
  scand: TeleconnectionIndex | null;
  mjo: MJOData | null;
  fetchedAt: number;
}

interface CachedSnapshot {
  data: TeleconnectionSnapshot;
  fetchedAt: number;
}

function readCache(key: string): CachedSnapshot | null {
  try {
    const saved = JSON.parse(localStorage.getItem(key) ?? 'null');
    if (!saved || !isFiniteNumber(saved.fetchedAt) || saved.fetchedAt < 0 || saved.fetchedAt > Date.now() + 60_000) return null;
    const data = saved.data;
    if (!isRecord(data) || data.fetchedAt !== saved.fetchedAt) return null;
    const indices = ['nao', 'ao', 'enso', 'pna', 'ea', 'scand'];
    if (!indices.every((slug) => data[slug] === null || validIndex(data[slug]))) return null;
    if (data.mjo !== null && !validMJO(data.mjo)) return null;
    if (!indices.some((slug) => data[slug] !== null) && data.mjo === null) return null;
    return saved;
  } catch { return null; }
}

function writeCache(key: string, data: TeleconnectionSnapshot): void {
  try { localStorage.setItem(key, JSON.stringify({ data, fetchedAt: data.fetchedAt })); } catch { /* storage full */ }
}

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null && !Array.isArray(value);
}

function isFiniteNumber(value: unknown): value is number {
  return typeof value === 'number' && Number.isFinite(value);
}

function validIndex(value: unknown): value is TeleconnectionIndex {
  return isRecord(value) && isFiniteNumber(value.value) && typeof value.date === 'string'
    && typeof value.slug === 'string' && typeof value.name === 'string';
}

function validMJO(value: unknown): value is MJOData {
  return isRecord(value) && isFiniteNumber(value.phase) && Number.isInteger(value.phase)
    && value.phase >= 1 && value.phase <= 8 && isFiniteNumber(value.amplitude) && value.amplitude >= 0
    && isFiniteNumber(value.rmm1) && isFiniteNumber(value.rmm2) && typeof value.date === 'string';
}

/** Extract an index from the flat API response object (keyed by slug). */
function extractIndex(body: unknown, slug: string): TeleconnectionIndex | null {
  const item = isRecord(body) ? body[slug] : null;
  if (!isRecord(item) || !isFiniteNumber(item.value)) return null;
  return {
    slug,
    name: typeof item.name === 'string' ? item.name : slug.toUpperCase(),
    value: item.value,
    date: typeof item.updated === 'string' ? item.updated : '',
    // The API's phase describes the sign, not the direction of change.
  };
}

async function fetchJSON(url: string): Promise<unknown> {
  const response = await fetch(url, { signal: AbortSignal.timeout(12_000) });
  if (!response.ok) throw new Error('Teleconnection data unavailable');
  return response.json();
}

let pending: Promise<TeleconnectionSnapshot> | null = null;

export function fetchTeleconnections(force = false): Promise<TeleconnectionSnapshot> {
  const cached = readCache(CACHE_KEY);
  if (!force && cached && Date.now() - cached.fetchedAt < CACHE_AGE) return Promise.resolve(cached.data);
  if (pending) return pending;

  pending = (async () => {
    const fetchedAt = Date.now();
    const snapshot: TeleconnectionSnapshot = {
      nao: null, ao: null, enso: null, pna: null, ea: null, scand: null, mjo: null, fetchedAt,
    };

    // One unavailable endpoint must not discard the other endpoint's data.
    const [main, mjo] = await Promise.allSettled([
      fetchJSON(API_BASE),
      fetchJSON(`${API_BASE}/mjo`),
    ]);

    if (main.status === 'fulfilled') {
      for (const slug of ['nao', 'ao', 'enso', 'pna', 'ea', 'scand'] as const) {
        snapshot[slug] = extractIndex(main.value, slug);
      }
    }

    if (mjo.status === 'fulfilled' && isRecord(mjo.value)) {
      const latest = mjo.value.latest ?? mjo.value;
      if (validMJO(latest)) snapshot.mjo = latest;
    }

    const hasIndices = snapshot.nao || snapshot.ao || snapshot.enso || snapshot.pna || snapshot.ea || snapshot.scand;
    if (!hasIndices && !snapshot.mjo) {
      throw new Error('Teleconnection data unavailable');
    }
    // Keep the previous successful cache when a refresh is incomplete.
    if (hasIndices && snapshot.mjo) writeCache(CACHE_KEY, snapshot);
    return snapshot;
  })().finally(() => { pending = null; });

  return pending;
}

export function readTeleconnections(): TeleconnectionSnapshot | null {
  return readCache(CACHE_KEY)?.data ?? null;
}

  const results: ActiveShower[] = [];

  for (const shower of METEOR_SHOWERS) {
    const activeStart = parseMMDD(shower.activeFrom, year);
    const activeEnd = parseMMDD(shower.activeTo, year);
    const peakStart = parseMMDD(shower.peakFrom, year);
    const peakEnd = parseMMDD(shower.peakTo, year);

    // Handle year-wrapping (e.g. Dec-Jan for Quadrantids)
    if (activeEnd < activeStart) activeEnd.setFullYear(year + 1);
    if (peakEnd < peakStart) peakEnd.setFullYear(year + 1);

    const isActive = now >= activeStart && now <= activeEnd;
    const daysToPeakStart = Math.round((peakStart.getTime() - now.getTime()) / 86400_000);
    const daysToPeakEnd = Math.round((peakEnd.getTime() - now.getTime()) / 86400_000);
    const isPeak = daysToPeakStart <= 0 && daysToPeakEnd >= 0;
    const isApproaching = daysToPeakStart > 0 && daysToPeakStart <= 30;

    if (!isActive && !isApproaching) continue;

    const peakDate = peakStart;
    const peakLabel = `${peakDate.toLocaleDateString('en-GB', { day: 'numeric', month: 'short' })}${
      shower.peakFrom !== shower.peakTo ? '–' + parseMMDD(shower.peakTo, year).toLocaleDateString('en-GB', { day: 'numeric', month: 'short' }) : ''
    }`;

    let status: ActiveShower['status'];
    if (isPeak) status = 'Peak tonight';
    else if (isActive && daysToPeakStart > 0 && daysToPeakStart <= 3) status = 'Peak approaching';
    else if (isActive) status = 'Active now';
    else status = 'Coming soon';

    results.push({
      ...shower,
      isPeak,
      daysToPeak: daysToPeakStart,
      peakLabel,
      status,
    });
  }

  // Sort: peak tonight first, then by days to peak
  results.sort((a, b) => {
    if (a.isPeak && !b.isPeak) return -1;
    if (!a.isPeak && b.isPeak) return 1;
    return a.daysToPeak - b.daysToPeak;
  });

  return results;
}

/** Get annual sky events for the current month (±15 days) */
export function getMonthlyEvents(dateStr: string): SkyEvent[] {
  const now = new Date(`${dateStr}T12:00:00Z`);
  const month = now.getMonth() + 1;
  return ANNUAL_EVENTS.filter((e) => {
    // Within ±1 month
    const diff = e.month - month;
    return diff === 0 || diff === 1 || diff === -11; // current or next month (wrapping Dec→Jan)
  });
}

import type { CachedRun } from '../types';
import type { VariableId } from '../config/models';
import { DETERMINISTIC_MODELS, VARIABLES } from '../config/models';
import { calcChange, calcFlipFlop, calcPersistence } from '../metrics/metrics';
import { getModelRuns } from '../metrics/utils';
import { format } from 'date-fns';

/** Material change thresholds (in display units) */
const MATERIAL: Record<string, number> = {
  temperature_2m: 1,
  precipitation: 1,     // mm in 6h
  wind_gusts_10m: 4,    // mph
  wind_speed_10m: 4,    // mph
  pressure_msl: 2,      // hPa
};

const VAR_LABELS: Record<string, { up: string; down: string }> = {
  temperature_2m: { up: 'warmer', down: 'cooler' },
  precipitation: { up: 'wetter', down: 'drier' },
  wind_gusts_10m: { up: 'gustier', down: 'calmer' },
  wind_speed_10m: { up: 'windier', down: 'calmer' },
  pressure_msl: { up: 'higher pressure', down: 'lower pressure' },
};

function getUnit(variable: VariableId): string {
  return VARIABLES.find((v) => v.id === variable)?.unit ?? '';
}

function formatValidTime(validTime: string): string {
  return format(new Date(validTime + 'Z'), 'EEEE HH:mm');
}

export interface Commentary {
  text: string;
  validTime: string;
  variable: VariableId;
}

/** Generate commentary for a variable at a specific valid time */
export function generateCommentary(
  allRuns: CachedRun[],
  variable: VariableId,
  validTime: string
): Commentary {
  const lines: string[] = [];
  const unit = getUnit(variable);
  const labels = VAR_LABELS[variable] ?? { up: 'rising', down: 'falling' };
  const materialThreshold = MATERIAL[variable] ?? 1;
  const timeLabel = formatValidTime(validTime);

  const trendingFamilies: string[] = [];
  const changelessFamilies: string[] = [];
  const missingFamilies: string[] = [];

  for (const model of DETERMINISTIC_MODELS) {
    if (model.isOverlay) continue;

    const runs = getModelRuns(allRuns, model.id, variable);
    if (runs.length === 0) {
      missingFamilies.push(model.label);
      continue;
    }

    const change = calcChange(runs, validTime);
    const flipFlop = calcFlipFlop(runs, validTime, variable);

    // Trending if material change, low flip-flops, and enough runs
    if (
      change.change24h !== null &&
      Math.abs(change.change24h) >= materialThreshold &&
      flipFlop.reversals <= 2 &&
      runs.length >= 4
    ) {
      const direction = change.change24h > 0 ? labels.up : labels.down;
      const sign = change.change24h > 0 ? '+' : '';
      lines.push(
        `${model.label} is trending ${direction} for ${timeLabel} (${sign}${change.change24h.toFixed(1)} ${unit} over 24h).`
      );
      trendingFamilies.push(model.label);
    } else if (flipFlop.reversals > 2 && runs.length >= 4) {
      lines.push(
        `${model.label} remains changeable for ${timeLabel} (${flipFlop.reversals} reversals in ${flipFlop.totalRuns} runs).`
      );
      changelessFamilies.push(model.label);
    }

    // Persistence for precip
    if (variable === 'precipitation' && runs.length >= 4) {
      const persist = calcPersistence(runs, validTime, 1, 8, 'gt');
      if (persist.total >= 4) {
        lines.push(
          `${persist.count} of ${persist.total} recent ${model.label} runs show >1 mm in the 6h to this time.`
        );
      }
    }
  }

  if (missingFamilies.length > 0) {
    lines.push(`No data from ${missingFamilies.join(', ')}.`);
  }

  if (lines.length === 0) {
    lines.push(`No material changes detected for ${timeLabel}.`);
  }

  return { text: lines.join(' '), validTime, variable };
}

/** Generate headline summaries across multiple valid times */
export function generateHeadlines(
  allRuns: CachedRun[],
  variable: VariableId,
  validTimes: string[]
): string[] {
  const headlines: string[] = [];

  for (const vt of validTimes) {
    const c = generateCommentary(allRuns, variable, vt);
    if (!c.text.includes('No material changes')) {
      headlines.push(c.text);
    }
  }

  return headlines;
}

interface InstallPrompt extends Event {
  prompt(): Promise<void>;
  userChoice: Promise<{ outcome: 'accepted' | 'dismissed' }>;
}

// Capture the browser's prompt before React mounts, including on cached visits.
let state: { prompt: InstallPrompt | null; error: boolean } = { prompt: null, error: false };
const listeners = new Set<() => void>();
const notify = () => listeners.forEach((listener) => listener());
const standalone = () => {
  try {
    return window.matchMedia('(display-mode: standalone)').matches
      || (navigator as Navigator & { standalone?: boolean }).standalone === true;
  } catch { return false; }
};

window.addEventListener('beforeinstallprompt', (event) => {
  if (standalone()) return;
  event.preventDefault();
  state = { prompt: event as InstallPrompt, error: false };
  notify();
});
window.addEventListener('appinstalled', () => {
  state = { prompt: null, error: false };
  notify();
});

const subscribe = (listener: () => void) => {
  listeners.add(listener);
  return () => { listeners.delete(listener); };
};
const getState = () => state;
const serverState: typeof state = { prompt: null, error: false };
const getServerState = () => serverState;

export function InstallApp() {
  const { prompt, error } = useSyncExternalStore(subscribe, getState, getServerState);

  async function install() {
    if (!prompt || state.prompt !== prompt) return;
    // Each browser event can only be used once, even if installation is declined.
    const attempt = { prompt: null, error: false };
    state = attempt;
    notify();
    try {
      await prompt.prompt();
      await prompt.userChoice;
    } catch {
      if (state === attempt) {
        state = { prompt: null, error: true };
        notify();
      }
    }
  }

  if (standalone() || (!prompt && !error)) return null;
  return <div className="install-offer">
    {prompt ? <>
      <span>Keep your local forecast handy.</span>
      <button type="button" onClick={install}>Install weather app</button>
    </> : <p role="status">Installation couldn’t start. Refresh this page and try again.</p>}
  </div>;
}

/** Calendar date in Peterborough rather than the device's time zone. */
export function getLocalDate(date: Date): string {
  const parts = new Intl.DateTimeFormat('en-GB', {
    timeZone: 'Europe/London', year: 'numeric', month: '2-digit', day: '2-digit',
  }).formatToParts(date);
  const value = (type: string) => parts.find((part) => part.type === type)!.value;
  return `${value('year')}-${value('month')}-${value('day')}`;
}

/** UTC forecast hours belonging to a calendar day in Peterborough, including clock changes. */
export function getLocalDayHours(dateISO: string): string[] {
  const formatter = new Intl.DateTimeFormat('en-GB', {
    timeZone: 'Europe/London', year: 'numeric', month: '2-digit', day: '2-digit',
  });
  const midnightUTC = Date.parse(`${dateISO}T00:00:00Z`);
  const selectedDate = formatter.format(new Date(`${dateISO}T12:00:00Z`));
  const hours: string[] = [];
  // UK local midnight is either 23:00 UTC on the previous day or 00:00 UTC.
  for (let offset = -1; offset < 25; offset++) {
    const time = new Date(midnightUTC + offset * 3600_000);
    if (formatter.format(time) === selectedDate) hours.push(time.toISOString().slice(0, 16));
  }
  return hours;
}

/** Today starts with the current hour; other selected days start at 09:00 local time. */
export function getHourlyForecastTimes(dateISO: string, now: Date): string[] {
  const isToday = dateISO === getLocalDate(now);
  const currentHour = Math.floor(now.getTime() / 3600_000) * 3600_000;
  return getLocalDayHours(dateISO).filter((time) => isToday
    ? Date.parse(`${time}Z`) >= currentHour
    : Number(toLocalTime(time, 'time').slice(0, 2)) >= 9);
}

/** Format a UTC ISO string to Europe/London local time */
export function toLocalTime(utcISO: string, formatStr: 'short' | 'long' | 'time' = 'short'): string {
  const date = new Date(utcISO.endsWith('Z') ? utcISO : utcISO + 'Z');
  const opts: Intl.DateTimeFormatOptions = { timeZone: 'Europe/London' };

  switch (formatStr) {
    case 'short':
      return date.toLocaleString('en-GB', { ...opts, weekday: 'short', day: 'numeric', month: 'short', hour: '2-digit', minute: '2-digit' });
    case 'long':
      return date.toLocaleString('en-GB', { ...opts, weekday: 'long', day: 'numeric', month: 'long', year: 'numeric', hour: '2-digit', minute: '2-digit' });
    case 'time':
      return date.toLocaleString('en-GB', { ...opts, hour: '2-digit', minute: '2-digit' });
  }
}

/** Format relative time ("2h ago") */
export function timeAgo(ms: number): string {
  const diff = Date.now() - ms;
  const mins = Math.floor(diff / 60000);
  if (mins < 1) return 'just now';
  if (mins < 60) return `${mins}m ago`;
  const hours = Math.floor(mins / 60);
  if (hours < 24) return `${hours}h ago`;
  const days = Math.floor(hours / 24);
  return `${days}d ago`;
}

const CHECKED_KEY = 'pb-weather:last-checked';
/** Re-check automatically when the app comes back to the foreground after this long. */
const AUTO_REFRESH_MS = 60 * 60_000;

function readLastChecked(): number | null {
  try {
    const value = Number(localStorage.getItem(CHECKED_KEY));
    return Number.isFinite(value) && value > 0 ? value : null;
  } catch { return null; }
}

export function useWeatherData() {
  const [state, setState] = useState<WeatherDataState>({
    runs: [],
    loading: true,
    progress: null,
    lastUpdated: readLastChecked(),
    error: null,
  });
  const inFlight = useRef(false);
  const lastChecked = useRef<number | null>(readLastChecked());

  const loadData = useCallback(async () => {
    if (inFlight.current) return;
    inFlight.current = true;
    setState((s) => ({ ...s, loading: true, error: null }));

    try {
      // Show saved data straight away
      const cachedRuns = await getAllCachedRuns();
      if (cachedRuns.length > 0) {
        setState((s) => ({
          ...s,
          runs: cachedRuns,
          lastUpdated: lastChecked.current ?? Math.max(...cachedRuns.map((r) => r.fetchedAt)),
        }));
      }

      const summary = await fetchAllDeterministicRuns(cachedRuns, (done, total) => {
        setState((s) => ({ ...s, progress: { done, total, phase: 'Fetching model runs' } }));
      });

      const reachedServer = summary.ok + summary.unavailable > 0;
      if (reachedServer) {
        lastChecked.current = Date.now();
        try { localStorage.setItem(CHECKED_KEY, String(lastChecked.current)); } catch { /* not critical */ }
      }

      const allRuns = summary.ok > 0 ? await getAllCachedRuns() : cachedRuns;
      const error = summary.requested === 0 || reachedServer ? null
        : allRuns.length ? "Couldn't reach the forecast service. Showing the forecast saved on this device."
        : "Couldn't reach the forecast service. Check your connection and try Refresh.";

      setState((s) => ({
        runs: allRuns,
        loading: false,
        progress: null,
        lastUpdated: lastChecked.current ?? s.lastUpdated,
        error,
      }));
    } catch (err) {
      setState((s) => ({
        ...s,
        loading: false,
        progress: null,
        error: err instanceof Error ? err.message : 'Failed to load data',
      }));
    } finally {
      inFlight.current = false;
    }
  }, []);

  useEffect(() => {
    // Leftovers from the retired wttr.in reference blend.
    try { localStorage.removeItem('pw_reference_forecast'); localStorage.removeItem('pw_reference_forecast:v2'); } catch { /* not critical */ }
    loadData();
    // A home-screen PWA can sit in the background for hours; catch up when it's reopened.
    const onVisible = () => {
      if (document.hidden) return;
      if (lastChecked.current === null || Date.now() - lastChecked.current > AUTO_REFRESH_MS) loadData();
    };
    document.addEventListener('visibilitychange', onVisible);
    return () => document.removeEventListener('visibilitychange', onVisible);
  }, [loadData]);

  return { ...state, refresh: loadData };
}

const mean = (values: number[]): number | null => values.length ? values.reduce((sum, value) => sum + value, 0) / values.length : null;
const max = (values: number[]): number | null => values.length ? Math.max(...values) : null;
const min = (values: number[]): number | null => values.length ? Math.min(...values) : null;
const available = (values: (number | null)[]): number[] => values.filter((value): value is number => value !== null);
// Weather codes are categories; their numeric mean would describe a different condition.
const mostFrequentCode = (values: number[]): number | null => {
  const counts = new Map<number, number>();
  for (const value of values) {
    if (Number.isInteger(value) && value >= 0 && value <= 99) counts.set(value, (counts.get(value) ?? 0) + 1);
  }
  return [...counts].sort((a, b) => b[1] - a[1])[0]?.[0] ?? null;
};
const degrees = (value: number) => `${Math.round(value)}°C`;
const periodName = (day: SummaryDay) => day.label !== 'Today' ? day.label : day.remainingToday ? 'the rest of today' : 'today';
const joinDays = (days: SummaryDay[]) => new Intl.ListFormat('en-GB', { style: 'long', type: 'conjunction' }).format(days.map(periodName));
const fullRain = (day: SummaryDay) => day.rainHours === day.expectedHours && day.expectedHours > 0;

function rainTiming(hours: Hour[]): string | null {
  const wet = hours.filter((hour) => hour.rain !== null && hour.rain >= 0.1);
  if (!wet.length) return null;
  const periods = [
    { label: 'overnight (00:00–06:00)', start: 0, end: 6 },
    { label: 'in the morning (06:00–12:00)', start: 6, end: 12 },
    { label: 'in the afternoon (12:00–18:00)', start: 12, end: 18 },
    { label: 'in the evening (18:00–00:00)', start: 18, end: 24 },
  ].map((period) => ({ ...period, total: wet.reduce((sum, hour) => {
    const localHour = Number(toLocalTime(hour.time, 'time').slice(0, 2));
    return sum + (localHour >= period.start && localHour < period.end ? hour.rain! : 0);
  }, 0) })).filter((period) => period.total > 0).sort((a, b) => b.total - a.total);
  if (periods.length >= 3) return 'Rain is spread across several parts of the day.';
  return `Rain is mainly indicated ${periods[0].label}${periods.length === 2 ? ` and ${periods[1].label}` : ''}.`;
}

function describeDay(day: SummaryDay): string[] {
  const lines: string[] = [];
  if (day.high !== null && day.low !== null) {
    lines.push(`Temperatures range from ${degrees(day.low)} to ${degrees(day.high)}${day.temperatureHours < day.expectedHours ? ' across the available hours' : ''}.`);
  } else lines.push('Temperature data is unavailable.');

  if (day.rainMm === null) lines.push('Rainfall data is unavailable; this is not a dry-weather signal.');
  else {
    const amount = day.rainMm.toFixed(1);
    lines.push(fullRain(day)
      ? day.rainMm < 0.1 ? 'Little or no rain is indicated.' : `Around ${amount} mm of rain is indicated. ${day.rainTiming ?? 'Only small amounts are indicated in individual hours.'}`
      : `The available hours show ${amount} mm of rain, but coverage is incomplete (${day.rainHours} of ${day.expectedHours} hours).${day.rainTiming ? ` ${day.rainTiming}` : ''}`);
  }

  if (day.windMph !== null) lines.push(`Winds average around ${Math.round(day.windMph)} mph across the available hours${day.gustMph !== null ? `, with peak gusts near ${Math.round(day.gustMph)} mph${day.gustTime ? ` around ${day.gustTime}` : ''}` : '; gust data is unavailable'}.`);
  else if (day.gustMph !== null) lines.push(`Peak gusts are around ${Math.round(day.gustMph)} mph${day.gustTime ? ` near ${day.gustTime}` : ''}; average wind data is unavailable.`);
  else lines.push('Wind and gust data is unavailable.');

  if (day.snowCm !== null && day.snowCm > 0) lines.push(`The models indicate ${day.snowCm < 0.1 ? 'a trace of snowfall' : `around ${day.snowCm.toFixed(1)} cm of snowfall`} across the available hours; settling is uncertain.`);
  if (day.low !== null && day.low <= 0) lines.push('Air temperatures reach freezing or below during the forecast period.');
  if (day.temperatureSpread !== null) lines.push(`${day.modelCount} temperature models contribute, with a largest same-hour difference of ${day.temperatureSpread.toFixed(1)}°C.`);
  else if (day.modelCount === 1) lines.push('Only one temperature model is available, so model agreement cannot be assessed.');
  return lines;
}

export interface BriefingOptions {
  /** Include today's earlier hours (the home screen's day cards) rather than only what's left. */
  wholeToday?: boolean;
}

/** Generate a briefing from the latest non-overlay runs, using Peterborough's calendar days. */
export function buildForecastBriefing(runs: CachedRun[], now: Date, options: BriefingOptions = {}): ForecastBriefing {
  const models = DETERMINISTIC_MODELS.filter((model) => !model.isOverlay);
  const modelIds = new Set(models.map((model) => model.id));
  const latest = new Map<string, CachedRun>();
  for (const run of runs) {
    if (!modelIds.has(run.modelId)) continue;
    const key = `${run.modelId}:${run.variable}`;
    if (!latest.has(key) || latest.get(key)!.runTime < run.runTime) latest.set(key, run);
  }
  // Shared with the home screen: newest run per model, gaps filled from the previous run, winds in mph.
  const latestIndex = buildLatestIndex(runs);
  const valuesAt = (variable: VariableId, time: string): number[] => models.flatMap((model) => {
    const value = latestIndex.value(model.id, variable, time);
    if (value === null) return [];
    if (variable !== 'temperature_2m' && value < 0) return [];
    return [value];
  });
  const today = getLocalDate(now);
  const currentHour = Math.floor(now.getTime() / 3600_000) * 3600_000;
  const days: SummaryDay[] = Array.from({ length: 7 }, (_, index) => {
    const date = new Date(Date.parse(`${today}T12:00:00Z`) + index * 86400_000).toISOString().slice(0, 10);
    const times = getLocalDayHours(date).filter((time) => index > 0 || options.wholeToday || Date.parse(`${time}Z`) >= currentHour);
    const hours: Hour[] = times.map((time) => {
      const temperatures = valuesAt('temperature_2m', time);
      return {
        time, weatherCode: mostFrequentCode(valuesAt('weather_code', time)), temperature: mean(temperatures), rain: mean(valuesAt('precipitation', time)),
        snow: mean(valuesAt('snowfall', time)), wind: mean(valuesAt('wind_speed_10m', time)),
        gust: mean(valuesAt('wind_gusts_10m', time)), temperatureModels: temperatures.length,
        temperatureSpread: temperatures.length > 1 ? Math.max(...temperatures) - Math.min(...temperatures) : null,
      };
    });
    const rain = available(hours.map((hour) => hour.rain));
    const snow = available(hours.map((hour) => hour.snow));
    const gust = max(available(hours.map((hour) => hour.gust)));
    const peakGustHours = hours.filter((hour) => gust !== null && hour.gust === gust);
    const day: SummaryDay = {
      date, label: index === 0 ? 'Today' : new Date(`${date}T12:00:00Z`).toLocaleDateString('en-GB', { weekday: 'long', timeZone: 'Europe/London' }),
      remainingToday: index === 0 && !options.wholeToday, expectedHours: times.length,
      temperatureHours: hours.filter((hour) => hour.temperature !== null).length, rainHours: rain.length,
      weatherCode: mostFrequentCode(available(hours.map((hour) => hour.weatherCode))),
      high: max(available(hours.map((hour) => hour.temperature))), low: min(available(hours.map((hour) => hour.temperature))),
      rainMm: rain.length ? rain.reduce((sum, value) => sum + value, 0) : null,
      snowCm: snow.length ? snow.reduce((sum, value) => sum + value, 0) : null,
      windMph: mean(available(hours.map((hour) => hour.wind))), gustMph: gust,
      gustTime: peakGustHours.length === 1 ? toLocalTime(peakGustHours[0].time, 'time') : null,
      rainTiming: rainTiming(hours), modelCount: Math.max(0, ...hours.map((hour) => hour.temperatureModels)),
      temperatureSpread: max(available(hours.map((hour) => hour.temperatureSpread))), paragraphs: [],
    };
    day.paragraphs = describeDay(day);
    return day;
  });
  const withTemp = days.filter((day) => day.high !== null);
  const completeTemp = withTemp.filter((day) => day.temperatureHours === day.expectedHours);
  const wet = days.filter((day) => day.rainMm !== null && day.rainMm >= 0.5);
  const completeRain = days.filter(fullRain);
  const warmest = [...completeTemp].sort((a, b) => b.high! - a.high!)[0];
  const coldest = [...completeTemp].sort((a, b) => a.low! - b.low!)[0];
  const wettest = [...completeRain].sort((a, b) => b.rainMm! - a.rainMm!)[0];
  const windiest = [...days].filter((day) => day.gustMph !== null).sort((a, b) => b.gustMph! - a.gustMph!)[0];
  const freezing = days.filter((day) => day.low !== null && day.low <= 0);
  const snowy = days.filter((day) => day.snowCm !== null && day.snowCm > 0);
  const hasData = days.some((day) => day.weatherCode !== null || day.high !== null || day.rainMm !== null || day.windMph !== null || day.gustMph !== null || day.snowCm !== null);

  let temperature = withTemp.length
    ? `Forecast highs range from ${degrees(Math.min(...withTemp.map((day) => day.high!)))} to ${degrees(Math.max(...withTemp.map((day) => day.high!)))} across the available days.`
    : 'Temperature data is unavailable, so a temperature trend cannot be assessed.';
  if (warmest && coldest) temperature += ` The highest forecast temperature is ${degrees(warmest.high!)} for ${periodName(warmest)}; the lowest is ${degrees(coldest.low!)} for ${periodName(coldest)}.`;
  if (completeTemp.length < 7 && withTemp.length) temperature += ' Temperature coverage is incomplete; the full range may be wider.';
  const early = mean(days.slice(1, 3).filter((day) => day.temperatureHours === day.expectedHours).map((day) => day.high!));
  const late = mean(days.slice(4, 7).filter((day) => day.temperatureHours === day.expectedHours).map((day) => day.high!));
  if (early !== null && late !== null) temperature += late - early >= 2
    ? ' The later days look warmer than the next couple of days.'
    : early - late >= 2 ? ' The later days look cooler than the next couple of days.' : ' Daytime highs look broadly similar between the next couple of days and later in the week.';

  let rainfall = completeRain.length === 7
    ? `${wet.length} of the next seven forecast periods ${wet.length === 1 ? 'has' : 'have'} at least 0.5 mm of rain indicated, totalling around ${days.reduce((sum, day) => sum + day.rainMm!, 0).toFixed(1)} mm.`
    : `Rainfall coverage is complete for ${completeRain.length} of the seven forecast periods. Missing hours cannot be treated as dry, so a full-week total is not available.`;
  if (wettest && wettest.rainMm! >= 0.5) rainfall += ` The wettest fully covered period is ${periodName(wettest)}, at around ${wettest.rainMm!.toFixed(1)} mm. ${wettest.rainTiming ?? ''}`;
  const dry = completeRain.filter((day) => day.rainMm! < 0.1);
  if (dry.length) rainfall += ` Little or no rain is indicated for ${joinDays(dry)}.`;

  const winds = available(days.map((day) => day.windMph));
  let wind = winds.length ? `Average winds range from around ${Math.round(Math.min(...winds))} to ${Math.round(Math.max(...winds))} mph across the available periods.` : 'Average wind data is unavailable.';
  wind += windiest ? ` The strongest forecast gusts are near ${Math.round(windiest.gustMph!)} mph for ${periodName(windiest)}${windiest.gustTime ? ` around ${windiest.gustTime}` : ''}.` : ' Gust data is unavailable.';
  const gusty = days.filter((day) => day.gustMph !== null && day.gustMph >= 30);
  if (gusty.length) wind += ` Gusts of 30 mph or more are indicated for ${joinDays(gusty)}.`;

  const cold = [
    freezing.length ? `Freezing or below-freezing air temperatures are indicated for ${joinDays(freezing)}.` : '',
    snowy.length ? `A snowfall signal appears for ${joinDays(snowy)}. Forecast snowfall does not guarantee snow will settle.` : '',
  ].filter(Boolean).join(' ');

  const compared = days.filter((day) => day.temperatureSpread !== null);
  const divergent = compared.filter((day) => day.temperatureSpread! >= 4);
  let agreement = compared.length ? `Temperature model agreement can be compared on ${compared.length} of the seven periods.` : 'There are not enough overlapping temperature forecasts to compare model agreement.';
  if (divergent.length) agreement += ` The models differ by at least 4°C at the same hour for ${joinDays(divergent)}, so the exact temperatures are less settled.`;
  else if (compared.length) agreement += ' Where multiple models overlap, their same-hour temperature differences stay below 4°C.';
  agreement += ' This measures model spread, not the probability of a forecast being correct. Rain timing and other details can change, especially further ahead.';
  const usedRuns = [...latest.values()].filter((run) => ['temperature_2m', 'precipitation', 'snowfall', 'wind_speed_10m', 'wind_gusts_10m'].includes(run.variable)
    && run.times.some((time, index) => Date.parse(`${time}Z`) >= currentHour && typeof run.values[index] === 'number' && Number.isFinite(run.values[index])));
  const latestRun = usedRuns.map((run) => run.runTime).sort().at(-1) ?? null;
  const stale = latestRun !== null && now.getTime() - Date.parse(`${latestRun}Z`) > 24 * 3600_000;
  const overview = !hasData ? 'A detailed briefing will appear once forecast data is available.'
    : `${temperature.split('. ')[0].replace(/\.$/, '')}. ${completeRain.length === 7 ? wet.length ? `Rain is indicated in ${wet.length} of the seven forecast periods.` : 'Very little rain is indicated across the seven forecast periods.' : 'Rainfall coverage is incomplete, so check the daily detail before planning.'}${windiest ? ` Peak gusts are around ${Math.round(windiest.gustMph!)} mph for ${periodName(windiest)}.` : ''}`;
  return { days, hasData, overview, temperature, rainfall, wind, cold, agreement, latestRun, stale };

export function conditionForCode(code: number | null, isDay: number | null): HourlyCondition {
  if (code === null || !Number.isInteger(code)) return UNAVAILABLE;
  if (code === 0 || code === 1) {
    if (isDay === 1) return { label: code === 0 ? 'Sunny' : 'Mainly sunny', icon: 'sun' };
    if (isDay === 0) return { label: code === 0 ? 'Clear night' : 'Mainly clear night', icon: 'moon' };
    return { label: code === 0 ? 'Clear sky' : 'Mainly clear', icon: null };
  }
  if (code === 2) {
    return { label: 'Partly cloudy', icon: isDay === 1 ? 'partly-cloudy' : isDay === 0 ? 'cloud-moon' : 'cloud' };
  }
  return CONDITIONS[code] ?? UNAVAILABLE;
}

/** Weather codes are categories: use the most common code, never their numeric average. */
export function getHourlyCondition(allRuns: CachedRun[], validTime: string): HourlyCondition {
  const votes = new Map<number, { count: number; condition: HourlyCondition }>();
  for (const model of DETERMINISTIC_MODELS) {
    if (model.isOverlay) continue;
    const modelRuns = getModelRuns(allRuns, model.id, 'weather_code');
    const latest = modelRuns.at(-1);
    if (!latest) continue;
    const code = getValueAtTime(latest, validTime);
    const daylight = allRuns.find((run) => run.modelId === model.id && run.variable === 'is_day' && run.runTime === latest.runTime);
    const condition = conditionForCode(code, daylight ? getValueAtTime(daylight, validTime) : null);
    if (condition === UNAVAILABLE || code === null) continue;
    const existing = votes.get(code);
    if (existing) existing.count++;
    else votes.set(code, { count: 1, condition });
  }
  // Stable ordering gives tied codes the configured model order as a consistent fallback.
  return [...votes.values()].sort((a, b) => b.count - a.count)[0]?.condition ?? UNAVAILABLE;
}

import type { CachedRun, ChangeMetric, FlipFlopMetric, PersistenceMetric } from '../types';
import type { VariableId } from '../config/models';
import { MS_PER_HOUR } from '../config/models';
import { getTrendValue, parseUtc, sortByRunTime } from './utils';

const FLAT_THRESHOLDS: Record<string, number> = {
  temperature_2m: 0.2,
  precipitation: 0.2,   // 6h total
  wind_gusts_10m: 0.5,  // mph, 6h max
  wind_speed_10m: 0.5,
  pressure_msl: 0.3,
};

/** Find the nearest run to a target age (in hours) within 3h tolerance */
function findRunByAge(runs: CachedRun[], newestRunTime: number, targetAgeH: number): CachedRun | null {
  const targetTime = newestRunTime - targetAgeH * MS_PER_HOUR;
  let best: CachedRun | null = null;
  let bestDiff = Infinity;

  for (const run of runs) {
    const diff = Math.abs(parseUtc(run.runTime) - targetTime);
    if (diff < bestDiff && diff <= 3 * MS_PER_HOUR) {
      bestDiff = diff;
      best = run;
    }
  }
  return best;
}

/** Calculate change metrics for a set of runs at a valid time */
export function calcChange(
  runs: CachedRun[],
  validTime: string
): ChangeMetric {
  const sorted = sortByRunTime(runs);
  if (sorted.length < 2) return { changePrev: null, change24h: null, change48h: null };

  const newest = sorted[sorted.length - 1];
  const newestVal = getTrendValue(newest, validTime);
  if (newestVal === null) return { changePrev: null, change24h: null, change48h: null };

  const newestMs = parseUtc(newest.runTime);

  const prev = sorted[sorted.length - 2];
  const prevVal = getTrendValue(prev, validTime);
  const changePrev = prevVal !== null ? newestVal - prevVal : null;

  const run24 = findRunByAge(sorted, newestMs, 24);
  const val24 = run24 ? getTrendValue(run24, validTime) : null;
  const change24h = val24 !== null ? newestVal - val24 : null;

  const run48 = findRunByAge(sorted, newestMs, 48);
  const val48 = run48 ? getTrendValue(run48, validTime) : null;
  const change48h = val48 !== null ? newestVal - val48 : null;

  return { changePrev, change24h, change48h };
}

/** Calculate flip-flop (direction reversals) */
export function calcFlipFlop(
  runs: CachedRun[],
  validTime: string,
  variable: VariableId
): FlipFlopMetric {
  const sorted = sortByRunTime(runs);
  const values: number[] = [];

  for (const run of sorted) {
    const v = getTrendValue(run, validTime);
    if (v !== null) values.push(v);
  }

  if (values.length < 3) return { reversals: 0, totalRuns: values.length };

  const threshold = FLAT_THRESHOLDS[variable] ?? 0.2;

  // Classify changes, removing flat steps
  const directions: ('up' | 'down')[] = [];
  for (let i = 1; i < values.length; i++) {
    const diff = values[i] - values[i - 1];
    if (Math.abs(diff) > threshold) {
      directions.push(diff > 0 ? 'up' : 'down');
    }
  }

  // Count reversals
  let reversals = 0;
  for (let i = 1; i < directions.length; i++) {
    if (directions[i] !== directions[i - 1]) reversals++;
  }

  return { reversals, totalRuns: values.length };
}

/** Calculate persistence — how many of the last N runs show an event */
export function calcPersistence(
  runs: CachedRun[],
  validTime: string,
  threshold: number,
  lastN: number = 8,
  comparator: 'gt' | 'lt' = 'gt'
): PersistenceMetric {
  const sorted = sortByRunTime(runs);
  const recent = sorted.slice(-lastN);
  let count = 0;
  let total = 0;

  for (const run of recent) {
    const v = getTrendValue(run, validTime);
    if (v !== null) {
      total++;
      if (comparator === 'gt' ? v > threshold : v < threshold) count++;
    }
  }

  return { count, total };
}

/** Weighted linear regression (trend slope) */
export function calcTrendSlope(
  runs: CachedRun[],
  validTime: string
): number | null {
  const sorted = sortByRunTime(runs);
  const points: { t: number; v: number }[] = [];

  for (const run of sorted) {
    const v = getTrendValue(run, validTime);
    if (v !== null) {
      points.push({ t: parseUtc(run.runTime), v });
    }
  }

  if (points.length < 4) return null;

  const spanH = (points[points.length - 1].t - points[0].t) / MS_PER_HOUR;
  if (spanH < 18) return null;

  const newestT = points[points.length - 1].t;
  const halfLifeMs = 36 * MS_PER_HOUR;

  // Weighted linear regression
  let sumW = 0, sumWx = 0, sumWy = 0, sumWxx = 0, sumWxy = 0;
  for (const p of points) {
    const ageMs = newestT - p.t;
    const w = Math.pow(0.5, ageMs / halfLifeMs);
    const x = (p.t - points[0].t) / (24 * MS_PER_HOUR); // x in days
    sumW += w;
    sumWx += w * x;
    sumWy += w * p.v;
    sumWxx += w * x * x;
    sumWxy += w * x * p.v;
  }

  const denom = sumW * sumWxx - sumWx * sumWx;
  if (Math.abs(denom) < 1e-10) return null;

  return (sumW * sumWxy - sumWx * sumWy) / denom; // units per day
}

export function temperatureSignal(anomaly: number | null): string {
  if (anomaly === null) return 'Temperature uncertain';
  return anomaly >= 0.5 ? 'Warmer than usual' : anomaly <= -0.5 ? 'Cooler than usual' : 'Near usual temperatures';
}

export function rainfallSignal(total: number | null, anomaly: number | null): string {
  if (total === null || anomaly === null || total - anomaly <= 0) return 'Rainfall uncertain';
  const ratio = anomaly / (total - anomaly);
  return ratio >= 0.2 && anomaly >= 2 ? 'Wetter than usual' : ratio <= -0.2 && anomaly <= -2 ? 'Drier than usual' : 'Near usual rainfall';
}

export function monthlySummary(weeks: OutlookWeek[], teleconnections?: TeleconnectionContext | null): string {
  if (weeks.length < 4) return 'Only part of the four-week outlook is available. Check the weekly signals below.';
  const temperature = weeks.every((week) => week.temperatureAnomaly !== null)
    ? temperatureSignal(weeks.reduce((sum, week) => sum + week.temperatureAnomaly!, 0) / weeks.length) : null;
  const rainfall = weeks.every((week) => week.rainfallMm !== null && week.rainfallAnomaly !== null)
    ? rainfallSignal(weeks.reduce((sum, week) => sum + week.rainfallMm!, 0), weeks.reduce((sum, week) => sum + week.rainfallAnomaly!, 0)) : null;
  const tempText = temperature === 'Warmer than usual' ? 'Temperatures lean warmer than usual overall.'
    : temperature === 'Cooler than usual' ? 'Temperatures lean cooler than usual overall.'
    : temperature === 'Near usual temperatures' ? 'Temperatures look close to the seasonal average overall.' : 'The temperature signal is incomplete.';
  const rainText = rainfall === 'Wetter than usual' ? 'Rainfall leans wetter than usual.'
    : rainfall === 'Drier than usual' ? 'Rainfall leans drier than usual.'
    : rainfall === 'Near usual rainfall' ? 'Rainfall looks close to the seasonal average.' : 'The rainfall signal is uncertain.';

  let tcText = '';
  // The temperature interpretation below is winter guidance, not an all-season rule.
  const winterOutlook = weeks.every((week) => [11, 12, 1, 2, 3].includes(Number(week.date.slice(5, 7))));
  if (winterOutlook && teleconnections?.mjoOutlook?.strong) {
    const mjo = teleconnections.mjoOutlook;
    // Check if MJO outlook aligns with or contradicts the model outlook
    const modelSaysCold = temperature === 'Cooler than usual';
    const modelSaysWarm = temperature === 'Warmer than usual';
    const mjoSaysCold = mjo.phase >= 6 && mjo.phase <= 8;
    const mjoSaysWarm = mjo.phase >= 2 && mjo.phase <= 3;

    if ((modelSaysCold && mjoSaysCold) || (modelSaysWarm && mjoSaysWarm)) {
      tcText = ` The active MJO in Phase ${mjo.phase} may support this temperature signal.`;
    } else if ((modelSaysCold && mjoSaysWarm) || (modelSaysWarm && mjoSaysCold)) {
      tcText = ` Note: the MJO in Phase ${mjo.phase} may work against this signal in the coming weeks.`;
    }
  }

  return `${tempText} ${rainText}${tcText}`;
}

function classifyRegime(nao: number, ao: number, scand: number | null): { id: RegimeId; confidence: 'high' | 'moderate' | 'low' } {
  const absNao = Math.abs(nao);
  const absAo = Math.abs(ao);

  // Strong positive NAO + positive AO → zonal
  if (nao > 0.8 && ao > 0.5) {
    return { id: 'zonal', confidence: absNao > 1.5 && absAo > 1.0 ? 'high' : 'moderate' };
  }

  // Strong negative NAO + negative AO → Greenland block
  if (nao < -0.8 && ao < -0.5) {
    return { id: 'greenland-block', confidence: absNao > 1.5 && absAo > 1.0 ? 'high' : 'moderate' };
  }

  // Positive Scand index with weak NAO → Scandinavian block
  if (scand !== null && scand > 0.8 && absNao < 1.0) {
    return { id: 'scandinavian-block', confidence: scand > 1.5 ? 'high' : 'moderate' };
  }

  // Weakly positive NAO, weak AO, no blocking → Azores ridge candidate
  if (nao > 0.3 && nao < 1.2 && ao > -0.3 && ao < 0.8 && (scand === null || scand < 0.5)) {
    return { id: 'azores-ridge', confidence: 'low' };
  }

  // Marginal signals
  if (absNao < 0.8 && absAo < 0.8) {
    return { id: 'transitional', confidence: 'low' };
  }

  // Weak but directional
  if (nao > 0.5) return { id: 'zonal', confidence: 'low' };
  if (nao < -0.5) return { id: 'greenland-block', confidence: 'low' };

  return { id: 'transitional', confidence: 'low' };
}

function mjoOutlook(mjo: MJOData): MJOOutlook {
  const strong = mjo.amplitude > 1.5;
  let naoImplication: string;
  let leadDays: string;

  if (!strong) {
    return {
      phase: mjo.phase,
      amplitude: mjo.amplitude,
      strong: false,
      naoImplication: 'Weak MJO signal — limited teleconnection influence on the UK.',
      confidenceNote: 'When the MJO is weak, its influence on mid-latitude weather patterns is minimal.',
      leadDays: '',
    };
  }

  // MJO phase → NAO implications (well-established relationships)
  switch (mjo.phase) {
    case 1:
      naoImplication = 'MJO Phase 1 can support a transition toward positive NAO.';
      leadDays = '10–15 days';
      break;
    case 2:
      naoImplication = 'MJO Phase 2 may favour positive NAO in winter, increasing the chance of Atlantic weather.';
      leadDays = '10–15 days';
      break;
    case 3:
      naoImplication = 'MJO Phase 3 may favour positive NAO and a more active Atlantic jet in winter.';
      leadDays = '10–14 days';
      break;
    case 4:
      naoImplication = 'MJO Phase 4 is transitional — NAO signal is mixed.';
      leadDays = '7–14 days';
      break;
    case 5:
      naoImplication = 'MJO Phase 5 can begin to weaken the polar vortex.';
      leadDays = '10–20 days';
      break;
    case 6:
      naoImplication = 'MJO Phase 6 may favour negative NAO in winter, increasing the chance of blocking and colder conditions.';
      leadDays = '10–14 days';
      break;
    case 7:
      naoImplication = 'MJO Phase 7 may favour negative NAO and blocking in winter; local conditions depend on the wider pattern.';
      leadDays = '10–14 days';
      break;
    case 8:
      naoImplication = 'MJO Phase 8 supports high pressure building — can prolong or trigger blocking.';
      leadDays = '7–14 days';
      break;
    default:
      naoImplication = 'MJO phase data unavailable.';
      leadDays = '';
  }

  const confidenceNote = mjo.amplitude > 2.0
    ? 'A larger MJO amplitude gives a clearer tropical signal, but its UK influence varies with season and the wider pattern.'
    : 'Moderate MJO amplitude — the signal is present but may be modulated by other factors.';

  return { phase: mjo.phase, amplitude: mjo.amplitude, strong, naoImplication, confidenceNote, leadDays };
}

function ensoLabel(value: number | null): string {
  if (value === null) return 'ENSO data unavailable';
  if (value >= 0.5) return 'El Niño';
  if (value <= -0.5) return 'La Niña';
  return 'ENSO-neutral';
}

function modelNotes(regime: RegimeClassification, mjo: MJOOutlook | null): string[] {
  const notes: string[] = [];

  if (regime.id === 'greenland-block' || regime.id === 'scandinavian-block') {
    notes.push('Blocking persistence can be difficult to forecast. Compare model agreement and successive runs.');
  }

  if (regime.id === 'transitional') {
    notes.push('Models are likely to show high flip-flop rates in transitional regimes.');
  }

  if (mjo?.strong && (mjo.phase === 6 || mjo.phase === 7)) {
    notes.push('An active MJO in phases 6–7 can influence winter circulation, but does not by itself predict stratospheric disruption.');
  }

  if (regime.id === 'zonal' && regime.confidence === 'high') {
    notes.push('Strong zonal flow is generally well-handled by all models — higher forecast confidence.');
  }

  return notes;
}

export function classifyTeleconnections(snapshot: TeleconnectionSnapshot): TeleconnectionContext {
  const naoVal = snapshot.nao?.value;
  const aoVal = snapshot.ao?.value;
  const scandVal = snapshot.scand?.value ?? null;
  const ensoVal = snapshot.enso?.value ?? null;

  const { id, confidence } = naoVal !== undefined && aoVal !== undefined
    ? classifyRegime(naoVal, aoVal, scandVal)
    : { id: 'unavailable' as const, confidence: 'low' as const };
  const regime: RegimeClassification = { ...REGIMES[id], confidence };

  const mjo = snapshot.mjo ? mjoOutlook(snapshot.mjo) : null;

  return {

import type { CachedRun, WindowValue } from '../types';
import type { VariableId } from '../config/models.ts';
import { DETERMINISTIC_MODELS, VARIABLES } from '../config/models.ts';

const MS_TO_MPH = 2.23694;
const HOUR = 3600_000;

/**
 * Open-Meteo times are UTC without a suffix ("2026-10-17T00:00"). `new Date()` reads that
 * form as device-local time, which is an hour out during BST, so always parse through here.
 */
export function parseUtc(iso: string): number {
  return Date.parse(iso.endsWith('Z') ? iso : `${iso}Z`);
}

const toIso = (ms: number) => new Date(ms).toISOString().slice(0, 16);

/** Convert m/s to mph if needed */
export function convertUnit(value: number | null, variable: VariableId): number | null {
  if (value === null) return null;
  const cfg = VARIABLES.find((v) => v.id === variable);
  if (cfg && 'convertFromMs' in cfg && cfg.convertFromMs) {
    return value * MS_TO_MPH;
  }
  return value;
}

/** Get the forecast value for a specific valid time from a run */
export function getValueAtTime(
  run: CachedRun,
  validTime: string
): number | null {
  const idx = run.times.indexOf(validTime);
  if (idx === -1) return null;
  return convertUnit(run.values[idx], run.variable);
}

/** Compute 6-hour precipitation total (sum) ending at windowEnd */
export function sixHourPrecipTotal(
  run: CachedRun,
  windowEndISO: string
): WindowValue {
  const endTime = parseUtc(windowEndISO);
  let sum = 0;
  let complete = true;
  let hasAny = false;

  for (let h = 5; h >= 0; h--) {
    const idx = run.times.indexOf(toIso(endTime - h * HOUR));
    if (idx === -1 || run.values[idx] === null) {
      complete = false;
    } else {
      sum += convertUnit(run.values[idx]!, run.variable) ?? 0;
      hasAny = true;
    }
  }

  return {
    windowEnd: windowEndISO,
    value: hasAny ? sum : null,
    complete,
  };
}

/** Compute 6-hour max gust ending at windowEnd */
export function sixHourMaxGust(
  run: CachedRun,
  windowEndISO: string
): WindowValue {
  const endTime = parseUtc(windowEndISO);
  let max: number | null = null;
  let complete = true;

  for (let h = 5; h >= 0; h--) {
    const idx = run.times.indexOf(toIso(endTime - h * HOUR));
    if (idx === -1 || run.values[idx] === null) {
      complete = false;
    } else {
      const v = convertUnit(run.values[idx]!, run.variable)!;
      if (max === null || v > max) max = v;
    }
  }

  return { windowEnd: windowEndISO, value: max, complete };
}

/**
 * The value the run-to-run trend tools compare. Hourly rain and gusts are too noisy to
 * track between runs, so these use the six hours ending at the valid time (the thresholds
 * in metrics.ts and commentary.ts are written for 6h totals). Incomplete windows return
 * null rather than an undercount.
 */
export function getTrendValue(run: CachedRun, validTime: string): number | null {
  if (run.variable === 'precipitation') {
    const w = sixHourPrecipTotal(run, validTime);
    return w.complete ? w.value : null;
  }
  if (run.variable === 'wind_gusts_10m') {
    const w = sixHourMaxGust(run, validTime);
    return w.complete ? w.value : null;
  }
  return getValueAtTime(run, validTime);
}

/** Generate 6-hour window end times (00, 06, 12, 18 UTC) covering a run's forecast */
export function generateWindows(run: CachedRun): string[] {
  if (run.times.length === 0) return [];
  const first = parseUtc(run.times[0]);
  const last = parseUtc(run.times[run.times.length - 1]);
  const windows: string[] = [];
  const startDay = new Date(first);
  startDay.setUTCHours(0, 0, 0, 0);

  for (let t = startDay.getTime(); t <= last; t += 6 * HOUR) {
    if (t >= first) windows.push(toIso(t));
  }
  return windows;
}

/** Sort runs by run time */
export function sortByRunTime(runs: CachedRun[]): CachedRun[] {
  return [...runs].sort((a, b) => a.runTime.localeCompare(b.runTime));
}

/** Get all runs for a model+variable, sorted by run time */
export function getModelRuns(
  allRuns: CachedRun[],
  modelId: string,
  variable: VariableId
): CachedRun[] {
  return sortByRunTime(
    allRuns.filter((r) => r.modelId === modelId && r.variable === variable)
  );
}

/**
 * Latest-run lookup built once per data load. Each model/variable maps to the newest run's
 * values by valid time; hours missing from the newest run (still publishing, or past its
 * horizon) fall back to the previous run so one late upload doesn't blank the forecast.
 */
export interface LatestIndex {
  value(modelId: string, variable: VariableId, validTime: string): number | null;
  /** Mean across non-overlay models' latest runs, in display units. */
  consensus(variable: VariableId, validTime: string): number | null;
}

export function buildLatestIndex(runs: CachedRun[]): LatestIndex {
  const grouped = new Map<string, CachedRun[]>();
  for (const run of runs) {
    const key = `${run.modelId}:${run.variable}`;
    const list = grouped.get(key);
    if (list) list.push(run); else grouped.set(key, [run]);
  }
  const index = new Map<string, Map<string, number>>();
  for (const [key, list] of grouped) {
    const [newest, previous] = sortByRunTime(list).reverse();
    const values = new Map<string, number>();
    for (const run of [previous, newest]) {
      if (!run) continue;
      run.times.forEach((time, i) => {
        const v = run.values[i];
        if (typeof v === 'number' && Number.isFinite(v)) values.set(time, v);
      });
    }
    index.set(key, values);
  }
  const models = DETERMINISTIC_MODELS.filter((m) => !m.isOverlay);

  const value = (modelId: string, variable: VariableId, validTime: string) => {
    const raw = index.get(`${modelId}:${variable}`)?.get(validTime);
    return raw === undefined ? null : convertUnit(raw, variable);
  };

  return {
    value,
    consensus(variable, validTime) {
      let sum = 0;
      let n = 0;
      for (const model of models) {
        const v = value(model.id, variable, validTime);
        if (v !== null) { sum += v; n++; }
      }
      return n ? sum / n : null;
    },
  };
}

function deriveCondition(rainMm: number, windMph: number): { condition: string } {
  const isWindy = windMph >= 25;
  const isBreezy = windMph >= 15;

  if (rainMm >= 8) {
    return isWindy
      ? { condition: 'Heavy rain & wind' }
      : { condition: 'Heavy rain' };
  }
  if (rainMm >= 2) {
    return isWindy
      ? { condition: 'Rain & wind' }
      : { condition: 'Rain' };
  }
  if (rainMm >= 0.5) {
    return isBreezy
      ? { condition: 'Showers & breezy' }
      : { condition: 'Showers' };
  }
  if (isWindy) return { condition: 'Windy' };
  if (isBreezy) return { condition: 'Breezy' };
  return { condition: 'Fair' };
}

function buildWeekSummary(days: DayForecast[]): string {
  const withData = days.filter((d) => d.high !== null);
  if (withData.length === 0) return 'Waiting for forecast data...';

  const lines: string[] = [];

  // Temperature overview
  const highs = withData.map((d) => d.high!);
  const avgHigh = highs.reduce((a, b) => a + b, 0) / highs.length;
  const maxHigh = Math.max(...highs);
  const minHigh = Math.min(...highs);

  const feel =
    avgHigh >= 25 ? 'a hot' :
    avgHigh >= 18 ? 'a warm' :
    avgHigh >= 12 ? 'a mild' :
    avgHigh >= 6 ? 'a cool' : 'a cold';

  if (maxHigh - minHigh >= 4) {
    lines.push(
      `Expect ${feel} week with highs between ${Math.round(minHigh)}°C and ${Math.round(maxHigh)}°C.`
    );
  } else {
    lines.push(
      `Expect ${feel} week with highs around ${Math.round(avgHigh)}°C.`
    );
  }

  // Rain overview
  const wetDays = withData.filter((d) => d.rainMm !== null && d.rainMm >= 0.5);
  const rainGaps = withData.some((d) => d.rainMm === null);
  if (wetDays.length === 0) {
    lines.push(rainGaps ? 'No rain showing on the days with full data.' : 'Looking dry all week.');
  } else if (wetDays.length <= 2) {
    const names = wetDays.map((d) => d.dayName).join(' and ');
    lines.push(`Some rain on ${names}.`);
  } else {
    lines.push(`Rain on ${wetDays.length} of the next ${withData.length} days.`);
  }

  // Wettest day callout
  const wettest = [...withData].sort((a, b) => (b.rainMm ?? 0) - (a.rainMm ?? 0))[0];
  if (wettest && wettest.rainMm !== null && wettest.rainMm >= 4) {
    lines.push(`${wettest.dayName} looks the wettest with around ${wettest.rainMm.toFixed(0)} mm.`);
  }

  // Wind callout
  const windyDays = withData.filter((d) => d.windMph !== null && d.windMph >= 20);
  if (windyDays.length > 0) {
    const names = windyDays.map((d) => d.dayName);
    if (windyDays.length === 1) {
      lines.push(`${names[0]} could be windy.`);
    } else {
      lines.push(`Breezy spells, especially ${names.slice(0, 2).join(' and ')}.`);
    }
  }

  return lines.join(' ');
}

  seeingRating: 'Excellent' | 'Good' | 'Fair' | 'Poor' | 'Unsuitable' | 'Unavailable';
  seeingScore: number | null;
  moonIllumination: number; // 0-100
  moonPhase: string;
  hourly: NightHour[];
}

interface NightHour {
  time: string;
  localTime: string;
  cloudLabel: string;
  cloudIcon: IconName;
  temp: number | null;
  wind: number | null;
  rain: number | null;
}

const SKY_MODELS = DETERMINISTIC_MODELS.filter((m) => !m.isOverlay);

/** Get the dominant weather code across models for a given time */
function consensusWeatherCode(index: LatestIndex, validTime: string): number | null {
  const votes = new Map<number, number>();
  for (const model of SKY_MODELS) {
    const code = index.value(model.id, 'weather_code', validTime);
    if (code !== null) votes.set(code, (votes.get(code) ?? 0) + 1);
  }
  let best: number | null = null;
  let bestCount = 0;
  for (const [code, count] of votes) {
    if (count > bestCount) { best = code; bestCount = count; }
  }
  return best;
}

/** Find sunset/sunrise from is_day transitions */
function findSunTransitions(index: LatestIndex, dateStr: string): { sunset: string | null; sunrise: string | null } {
  let sunset: string | null = null;
  let sunrise: string | null = null;

  // Look for the evening transition on dateStr and the morning transition on dateStr+1
  for (const model of SKY_MODELS) {
    const isDay = (t: string) => index.value(model.id, 'is_day', t);

    // Find sunset: last hour of daylight on dateStr
    for (let h = 12; h < 24; h++) {
      const t = `${dateStr}T${String(h).padStart(2, '0')}:00`;
      const next = `${dateStr}T${String(h + 1).padStart(2, '0')}:00`;
      const val = isDay(t);
      const nextVal = h < 23 ? isDay(next) : null;
      if (val === 1 && (nextVal === 0 || nextVal === null)) {
        sunset = t;
        break;
      }
    }

    // Find sunrise: first hour of daylight on the next day
    const nextDate = new Date(Date.parse(`${dateStr}T00:00:00Z`) + 86400_000).toISOString().slice(0, 10);
    for (let h = 4; h < 12; h++) {
      const t = `${nextDate}T${String(h).padStart(2, '0')}:00`;
      const val = isDay(t);
      if (val === 1) {
        sunrise = t;
        break;
      }
    }

    if (sunset || sunrise) break;
  }

  return { sunset, sunrise };
}

/** Calculate moon phase for a given date. Returns illumination 0–100 and phase name. */
function getMoonPhase(dateStr: string): { illumination: number; phase: string } {
  const date = new Date(`${dateStr}T00:00:00Z`);
  // Known new moon: 2000-01-06 18:14 UTC
  const knownNew = Date.UTC(2000, 0, 6, 18, 14);
  const synodicMonth = 29.53058770576;
  const daysSinceNew = (date.getTime() - knownNew) / 86400_000;
  const phase = ((daysSinceNew % synodicMonth) + synodicMonth) % synodicMonth;
  const normalised = phase / synodicMonth; // 0-1

  // Illumination: 0 at new/full boundaries mapped via cosine
  const illumination = Math.round((1 - Math.cos(normalised * 2 * Math.PI)) / 2 * 100);

  let name: string;
  if (normalised < 0.03 || normalised > 0.97) name = 'New Moon';
  else if (normalised < 0.22) name = 'Waxing Crescent';
  else if (normalised < 0.28) name = 'First Quarter';
  else if (normalised < 0.47) name = 'Waxing Gibbous';
  else if (normalised < 0.53) name = 'Full Moon';
  else if (normalised < 0.72) name = 'Waning Gibbous';
  else if (normalised < 0.78) name = 'Last Quarter';
  else name = 'Waning Crescent';

  return { illumination, phase: name };
}

/** Cloud description from WMO weather code */
function cloudFromCode(code: number | null): { label: string; icon: IconName } {
  if (code === null) return { label: 'Unknown', icon: 'cloud' };
  if (code === 0) return { label: 'Clear', icon: 'moon' };
  if (code === 1) return { label: 'Mostly clear', icon: 'cloud-moon' };
  if (code === 2) return { label: 'Partly cloudy', icon: 'cloud-moon' };
  if (code === 3) return { label: 'Overcast', icon: 'cloud' };
  if (code === 45 || code === 48) return { label: 'Fog', icon: 'fog' };
  if (code >= 51 && code <= 67) return { label: 'Rain', icon: 'rain' };
  if (code >= 71 && code <= 77) return { label: 'Snow', icon: 'snow' };
  if (code >= 80 && code <= 82) return { label: 'Showers', icon: 'rain' };
  if (code >= 85 && code <= 86) return { label: 'Snow showers', icon: 'snow' };
  if (code >= 95) return { label: 'Thunderstorm', icon: 'storm' };
  return { label: 'Cloudy', icon: 'cloud' };
}

/** Label from modelled total cloud cover; rain, snow, fog and storms keep their weather-code label. */
function cloudFromCover(cover: number, code: number | null): { label: string; icon: IconName } {
  if (code !== null && code > 3) return cloudFromCode(code);
  if (cover < 10) return { label: 'Clear', icon: 'moon' };
  if (cover < 35) return { label: 'Mostly clear', icon: 'cloud-moon' };
  if (cover < 70) return { label: 'Partly cloudy', icon: 'cloud-moon' };
  if (cover < 90) return { label: 'Mostly cloudy', icon: 'cloud' };
  return { label: 'Overcast', icon: 'cloud' };
}

/** Fallback cloud score from weather codes, for runs saved before cloud cover was fetched. */
function cloudScoreFromCode(code: number): number {
  if (code === 0) return 0;
  if (code === 1) return 15;
  if (code === 2) return 50;
  if (code === 3) return 90;
  if (code === 45 || code === 48) return 100; // fog
  if (code >= 51) return 95; // precipitation implies heavy cloud
  return 50;
}

function rateSeeingConditions(cloudScore: number, windMph: number | null, rainRisk: boolean): { rating: NightForecast['seeingRating']; score: number } {
  // Start from 100 and deduct
  let score = 100;

  // Cloud cover is the biggest factor
  score -= cloudScore * 0.7;

  // Wind causes atmospheric turbulence
  const wind = windMph ?? 0;
  if (wind > 25) score -= 25;
  else if (wind > 15) score -= 15;
  else if (wind > 8) score -= 5;

  // Rain is a dealbreaker
  if (rainRisk) score -= 20;

  score = Math.max(0, Math.min(100, Math.round(score)));

  let rating: NightForecast['seeingRating'];
  if (score >= 75) rating = 'Excellent';
  else if (score >= 55) rating = 'Good';
  else if (score >= 35) rating = 'Fair';
  else if (score >= 15) rating = 'Poor';
  else rating = 'Unsuitable';

  return { rating, score };
}

      const date = new Date(todayUTC + d * 86400_000);

      const { sunset, sunrise } = findSunTransitions(index, dateStr);

      // Night hours: from sunset (or 17:00) to sunrise (or 07:00 next day)
      const nightStart = sunset ? parseInt(sunset.slice(11, 13)) + 1 : 17;
      const nightEnd = sunrise ? parseInt(sunrise.slice(11, 13)) : 7;

      const cloudScores: number[] = [];
      const winds: number[] = [];
      const gusts: number[] = [];
      let hasRain = false;
      const hourly: NightHour[] = [];

      // Evening hours (nightStart to 23)
      for (let h = nightStart; h < 24; h++) {
        const t = `${dateStr}T${String(h).padStart(2, '0')}:00`;
        const code = consensusWeatherCode(index, t);
        const cover = consensus('cloud_cover', t);
        const cloud = cover !== null ? cloudFromCover(cover, code) : cloudFromCode(code);
        if (cover !== null) cloudScores.push(cover);
        else if (code !== null) cloudScores.push(cloudScoreFromCode(code));

        const wind = consensus('wind_speed_10m', t);
        if (wind !== null) winds.push(wind);

        const gust = consensus('wind_gusts_10m', t);
        if (gust !== null) gusts.push(gust);

        const rain = consensus('precipitation', t);
        if (rain !== null && rain >= 0.1) hasRain = true;

        hourly.push({
          time: t,
          localTime: toLocalTime(t, 'time'),
          cloudLabel: cloud.label,
          cloudIcon: cloud.icon,
          temp: consensus('temperature_2m', t),
          wind,
          rain,
        });
      }

      // Early morning hours (00 to nightEnd)
      for (let h = 0; h < nightEnd; h++) {
        const t = `${nextDateStr}T${String(h).padStart(2, '0')}:00`;
        const code = consensusWeatherCode(index, t);
        const cover = consensus('cloud_cover', t);
        const cloud = cover !== null ? cloudFromCover(cover, code) : cloudFromCode(code);
        if (cover !== null) cloudScores.push(cover);
        else if (code !== null) cloudScores.push(cloudScoreFromCode(code));

        const wind = consensus('wind_speed_10m', t);
        if (wind !== null) winds.push(wind);

        const gust = consensus('wind_gusts_10m', t);
        if (gust !== null) gusts.push(gust);

        const rain = consensus('precipitation', t);
        if (rain !== null && rain >= 0.1) hasRain = true;

        hourly.push({
          time: t,
          localTime: toLocalTime(t, 'time'),
          cloudLabel: cloud.label,
          cloudIcon: cloud.icon,
          temp: consensus('temperature_2m', t),
          wind,
          rain,
        });
      }

      const avgCloud = cloudScores.length > 0 ? cloudScores.reduce((a, b) => a + b, 0) / cloudScores.length : null;
      const avgWind = winds.length > 0 ? winds.reduce((a, b) => a + b, 0) / winds.length : null;
      const maxGust = gusts.length > 0 ? Math.max(...gusts) : null;

      const { rating, score } = avgCloud !== null && avgWind !== null
        ? rateSeeingConditions(avgCloud, avgWind, hasRain)
        : { rating: 'Unavailable' as const, score: null };

      const shortDay = localDate.toLocaleDateString('en-GB', { weekday: 'short', timeZone: 'Europe/London' });
      const isTonight = d === 0;

      result.push({
        date: dateStr,
        label: isTonight ? 'Tonight' : `${dayName} night`,
        shortLabel: isTonight ? 'Tonight' : shortDay,
        sunset: sunset ? toLocalTime(sunset, 'time') : null,
        sunrise: sunrise ? toLocalTime(sunrise, 'time') : null,
        cloudScore: avgCloud === null ? null : Math.round(avgCloud),

        moonPhase: moon.phase,

    detail.current?.focus({ preventScroll: true });
    detail.current?.scrollIntoView({ behavior: window.matchMedia('(prefers-reduced-motion: reduce)').matches ? 'instant' : 'smooth', block: 'start' });

    {!hasData && <div className="nightsky-status" role="status">
      <WeatherIcon name="telescope" />
      <p>{loading ? 'Gathering your night sky forecast…' : hasError ? 'The forecast could not be loaded. Use Refresh above to try again.' : 'No forecast data is available. Use Refresh above to check again.'}<span>Moon phases are still available below.</span></p>
    </div>}
    {hasData && hasError && <p className="nightsky-status" role="status">The latest refresh failed. Showing the forecast already saved on this device.</p>}

function MoonDisc({ illumination, phase }: { illumination: number; phase: string }) {
  const id = useId();

interface Props {

function getValidTimes(): string[] {
  const now = new Date();
  const times: string[] = [];
  // Every 6 hours for the next 7 days
  const start = new Date(Date.UTC(now.getUTCFullYear(), now.getUTCMonth(), now.getUTCDate()));
  for (let h = 0; h <= 168; h += 6) {
    const t = new Date(start.getTime() + h * 3600_000);
    times.push(t.toISOString().slice(0, 16));
  }
  return times;
}

export function Trends({ runs }: Props) {
  const [variable, setVariable] = useState<VariableId>('temperature_2m');
  const [validTimeIdx, setValidTimeIdx] = useState(4); // ~24h ahead
  const [cycleFilter, setCycleFilter] = useState<number | null>(null);
  const validTimes = useMemo(() => getValidTimes(), []);
  const validTime = validTimes[validTimeIdx] ?? validTimes[0];

  const models = DETERMINISTIC_MODELS;
  const variableCfg = VARIABLES.find((v) => v.id === variable)!;

  // Chart data: x = run time, y = forecast value for the chosen valid time
  const chartData = useMemo(() => {
    // Collect all run times across all models
    const runTimeSet = new Set<string>();
    for (const model of models) {
      const modelRuns = getModelRuns(runs, model.id, variable);
      for (const run of modelRuns) {
        if (cycleFilter === null || new Date(run.runTime + 'Z').getUTCHours() === cycleFilter) {
          runTimeSet.add(run.runTime);
        }
      }
    }

    const runTimes = [...runTimeSet].sort();

    return runTimes.map((rt) => {
      const point: Record<string, string | number | null> = {
        runTime: rt,
        runTimeLabel: toLocalTime(rt, 'short'),
      };
      for (const model of models) {
        const modelRuns = getModelRuns(runs, model.id, variable);
        const run = modelRuns.find((r) => r.runTime === rt);
        point[model.id] = run ? getTrendValue(run, validTime) : null;
      }
      return point;
    });
  }, [runs, variable, validTime, models, cycleFilter]);

  // Metrics
  const metrics = useMemo(() => {
    return models
      .filter((m) => !m.isOverlay)
      .map((model) => {
        const modelRuns = getModelRuns(runs, model.id, variable);
        const change = calcChange(modelRuns, validTime);
        const flipFlop = calcFlipFlop(modelRuns, validTime, variable);
        const slope = calcTrendSlope(modelRuns, validTime);
        const persist = variable === 'precipitation'
          ? calcPersistence(modelRuns, validTime, 1, 8, 'gt')
          : null;

        return { model, change, flipFlop, slope, persist };
      });
  }, [runs, variable, validTime, models]);

  const commentary = useMemo(
    () => generateCommentary(runs, variable, validTime),
    [runs, variable, validTime]
  );

  return (
    <div className="view-trends">
      <VariableSelector selected={variable} onChange={setVariable} />

      <div className="valid-time-picker">
        <label>Valid time: </label>

          min={0}
          max={validTimes.length - 1}
          value={validTimeIdx}
          onChange={(e) => setValidTimeIdx(Number(e.target.value))}
        />
        <span className="valid-time-label">{toLocalTime(validTime, 'long')}{variable === 'precipitation' ? ' · 6h total to this time' : variable === 'wind_gusts_10m' ? ' · 6h max to this time' : ''}</span>
      </div>

      <div className="cycle-filter">
        <label>Cycle filter: </label>
        <select
          value={cycleFilter ?? 'all'}
          onChange={(e) => setCycleFilter(e.target.value === 'all' ? null : Number(e.target.value))}
        >
          <option value="all">All cycles</option>
          <option value="0">00Z</option>
          <option value="6">06Z</option>
          <option value="12">12Z</option>
          <option value="18">18Z</option>
        </select>
      </div>

      <div className="chart-container">
        <ResponsiveContainer width="100%" height={350}>
          <LineChart data={chartData} margin={{ top: 5, right: 20, bottom: 5, left: 0 }}>
            <CartesianGrid strokeDasharray="3 3" stroke="#333" />
            <XAxis
              dataKey="runTimeLabel"
              tick={{ fontSize: 11 }}
              interval="preserveStartEnd"
            />
            <YAxis
              tick={{ fontSize: 11 }}
              label={{ value: variableCfg.unit, angle: -90, position: 'insideLeft' }}
            />
            <Tooltip
              contentStyle={{ backgroundColor: '#1e1e2e', border: '1px solid #444' }}

              />
            ))}
